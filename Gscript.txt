// Safely converts a GAS date value to YYYY-MM-DD string.
// GAS auto-converts date cells in Sheets to JS Date objects, so we
// cannot assume .date is always a string — guard against both cases.
function gasDateToYMD(val) {
  if (!val) return '';
  if (val instanceof Date) {
    var y = val.getFullYear();
    var m = String(val.getMonth() + 1).padStart(2, '0');
    var d = String(val.getDate()).padStart(2, '0');
    return y + '-' + m + '-' + d;
  }
  return String(val).trim();
}

function generateToken() {
  return Utilities.getUuid().replace(/-/g, '') + Date.now().toString(36);
}

// Writes a new session row to the Sessions sheet. Sheets writes are
// synchronous and durable — unlike CacheService, there's no eventual-
// consistency window where a value written a moment ago isn't readable yet.
function storeSession(token, sessionData) {
  const sheet = getSheet(SHEET_SESSIONS);
  const now = new Date();
  const expiresAt = new Date(now.getTime() + SESSION_TTL_SEC * 1000);
  sheet.appendRow([token, JSON.stringify(sessionData), expiresAt, now]);
}

// Deletes a session row outright — real logout/revocation, as opposed to
// just letting a token sit until its sliding-expiration TTL naturally runs
// out. Silently no-ops on an empty/unknown token (matches validateSession's
// own tolerance for a bad token).
function deleteSessionByToken(token) {
  if (!token) return;
  const sheet = getSheet(SHEET_SESSIONS);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === token) { sheet.deleteRow(i + 1); return; }
  }
}

// Set by validateSession on every call so doGet/doPost can echo it back in
// the sessionExpired response — lets us see what happened straight from the
// Network tab instead of digging through the Executions log.
let LAST_VALIDATE_DEBUG = null;

// Session validation with sliding expiration, backed by the Sessions sheet.
function validateSession(token) {
  if (!token) { LAST_VALIDATE_DEBUG = { receivedToken: false }; return null; }
  const sheet = getSheet(SHEET_SESSIONS);
  const data = sheet.getDataRange().getValues();
  const now = new Date();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === token) {
      const expiresAt = new Date(data[i][2]);
      if (expiresAt < now) {
        LAST_VALIDATE_DEBUG = { receivedToken: true, token: token, tokenLength: token.length, foundInCache: false, reason: 'expired' };
        sheet.deleteRow(i + 1);
        return null;
      }
      // Sliding expiration: push the expiry out on every valid use.
      sheet.getRange(i + 1, 3).setValue(new Date(now.getTime() + SESSION_TTL_SEC * 1000));
      LAST_VALIDATE_DEBUG = { receivedToken: true, token: token, tokenLength: token.length, foundInCache: true };
      return JSON.parse(data[i][1]);
    }
  }
  LAST_VALIDATE_DEBUG = { receivedToken: true, token: token, tokenLength: token.length, foundInCache: false, reason: 'not found' };
  return null;
}

// ── MULTI-TENANCY HELPERS ──────────────────────────────────────────────────

function generateSalt() {
  return Utilities.getUuid().replace(/-/g, '');
}

// v1: a single SHA-256 round. Kept only so already-stored hashes (created
// before hashPasswordStrong existed) still verify — see verifyPassword.
// Never used for a NEW hash: too fast to be safe against brute force even
// with a per-user salt.
function hashPassword(password, salt) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, (salt || '') + '::' + (password || ''));
  return bytes.map(function(b) { return (b < 0 ? b + 256 : b).toString(16).padStart(2, '0'); }).join('');
}

// v2: SHA-256 iterated thousands of times (Apps Script has no native
// bcrypt/scrypt/Argon2) to deliberately slow down brute-force/GPU cracking
// of a stolen hash — a per-user salt alone only defeats rainbow tables, not
// raw guessing speed. Stored hashes are prefixed 'v2:' so verifyPassword can
// tell which algorithm produced them.
const HASH_V2_PREFIX = 'v2:';
const HASH_V2_ITERATIONS = 10000;
function hashPasswordStrong(password, salt) {
  var value = (salt || '') + '::' + (password || '');
  for (var i = 0; i < HASH_V2_ITERATIONS; i++) {
    var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, value);
    value = bytes.map(function(b) { return (b < 0 ? b + 256 : b).toString(16).padStart(2, '0'); }).join('');
  }
  return HASH_V2_PREFIX + value;
}

// Verifies a plaintext password against a stored hash of EITHER format and
// reports which one matched (or null on failure), so a caller can silently
// upgrade a v1 match to v2 the moment it's confirmed correct — no forced
// password reset, nothing visible to the user.
function verifyPassword(password, salt, storedHash) {
  storedHash = storedHash || '';
  if (storedHash.indexOf(HASH_V2_PREFIX) === 0) {
    return hashPasswordStrong(password, salt) === storedHash ? 'v2' : null;
  }
  return hashPassword(password, salt) === storedHash ? 'v1' : null;
}

// Applies the logged-in user's tenant to this request: every getSheet() call
// for business-data sheets (Invoices, Quotations, etc.) will now resolve to
// that tenant's own spreadsheet, and every print/email function will use
// that tenant's own company/bank details. Call this once, right after
// validateSession() succeeds, in both doGet and doPost.
function applyTenantContext(session) {
  if (!session) return;
  ACTIVE_SPREADSHEET_ID = session.spreadsheetId || null;
  if (session.company) COMPANY = session.company;
}

// Non-destructively brings an EXISTING sheet's header row up to date with a
// newer expected column list — appends any missing header as a new trailing
// column. Existing columns, their order, and all existing row data are never
// touched; older rows simply read back blank for the new column. This is
// what lets a schema change (e.g. this migration's Ship To / Logo / PAN
// columns) reach a live production spreadsheet without anyone hand-editing
// it first.
function ensureHeaders(sheet, expectedHeaders) {
  const lastCol = sheet.getLastColumn();
  const existing = lastCol > 0
    ? sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function(h) { return String(h).trim(); })
    : [];
  const missing = expectedHeaders.filter(function(h) { return existing.indexOf(h) === -1; });
  if (!missing.length) return;
  const startCol = lastCol + 1;
  const range = sheet.getRange(1, startCol, 1, missing.length);
  range.setValues([missing]);
  range.setFontWeight('bold').setBackground('#f1f5f9');
}

// Dynamic target sheet accessor with automated structural initialization

    sheet = ss.insertSheet(name);
    if (name === SHEET_INVOICES) {
      sheet.appendRow(INVOICE_HEADERS);
    } else if (name === SHEET_CHALLANS) {
      sheet.appendRow([
        'Challan No', 'Date', 'Ref Invoice No', 'Buyer Name', 'Buyer Phone', 'Buyer Email',
        'Buyer Address', 'Buyer GSTIN', 'Transport', 'Notes', 'Items (JSON)', 'Saved By', 'Timestamp'
      ]);
    } else if (name === SHEET_BUYERS) {
      sheet.appendRow([
        'Name', 'Address', 'GSTIN', 'State', 'State Code', 'Phone', 'Email', 'Saved By', 'Timestamp'
      ]);
    } else if (name === SHEET_ITEMS) {
      sheet.appendRow([
        'Name', 'Type', 'HSN', 'Unit', 'Rate', 'GST', 'Saved By', 'Timestamp'
      ]);
    } else if (name === SHEET_QUOTATIONS) {
      sheet.appendRow([
        'Quotation No', 'Date', 'Valid Until', 'Ref No', 'Buyer Name', 'Buyer Phone', 'Buyer Email',
        'Buyer Address', 'Buyer GSTIN', 'Buyer State', 'Sub Total', 'Tax Total', 'Grand Total',
        'Notes', 'Items (JSON)', 'Saved By', 'Timestamp'
      ]);
    } else if (name === SHEET_CREDIT_DEBIT_NOTES) {
      sheet.appendRow([
        'Note Type', 'Note No', 'Date', 'Against Invoice No', 'Against Invoice Date', 'Reason',
        'Buyer Name', 'Buyer Phone', 'Buyer Email', 'Buyer Address', 'Buyer GSTIN', 'Buyer State',
        'State Code', 'Sub Total', 'CGST Rate', 'CGST Amount', 'SGST Rate', 'SGST Amount',
        'IGST Rate', 'IGST Amount', 'Tax Amount', 'Grand Total', 'Items (JSON)', 'Saved By', 'Timestamp'
      ]);
    } else if (name === SHEET_AMC) {
      sheet.appendRow(AMC_HEADERS);
    } else if (name === SHEET_CERTIFICATES) {
      sheet.appendRow(CERTIFICATE_HEADERS);
    } else if (name === SHEET_TENANTS) {
      sheet.appendRow(TENANT_HEADERS);
    } else if (name === SHEET_USERS) {
      sheet.appendRow([
        'Username', 'Password Hash', 'Salt', 'Display Name', 'Tenant ID', 'Role',
        'Status', 'Created Date'

    ensureHeaders(sheet, AMC_HEADERS);
  } else if (name === SHEET_CERTIFICATES) {
    ensureHeaders(sheet, CERTIFICATE_HEADERS);
  }
  return sheet;
}

// ── GET ROUTER ───────────────────────────────────────────────────────────────

function doGet(e) {
  const p      = e.parameter || {};
  const action = p.action || '';
  const secret = p.secret || '';
  const token  = p.token  || '';

  try {
    // checkLogin moved to doPost — a password must never travel in a URL
    // query string (browser history, proxy/CDN logs, Referer leakage). See
    // the comment above doPost's checkLogin handler.

    // Super-admin-portal action — gated by the super-admin secret, not a
    // session token, since the portal manages tenants from outside any
    // tenant's own login. Must come before the session check below, or it
    // would always fail with "session expired" (no token is ever sent).
    if (action === 'listTenants') {
      if (!isSuperAdminSecretValid(secret)) {
        return jsonOut({ success: false, message: 'Unauthorized: invalid super-admin secret.' });
      }
      // Own try/catch: reachable by anyone who guesses this action name,
      // gated only by the secret above — an internal error shouldn't echo
      // raw exception text to a caller who hasn't even proven that secret
      // (this check already passed, but keep the response generic anyway
      // since it's still pre-session).
      try {
        return handleListTenants();
      } catch (err) {
        Logger.log('listTenants error: ' + err);
        return jsonOut({ success: false, message: 'Could not load tenants.' });
      }
    }
    if (action === 'listTenantUsers') {
      if (!isSuperAdminSecretValid(secret)) {
        return jsonOut({ success: false, message: 'Unauthorized: invalid super-admin secret.' });
      }
      try {
        return handleListTenantUsers(p.tenantId || '');
      } catch (err) {
        Logger.log('listTenantUsers error: ' + err);
        return jsonOut({ success: false, message: 'Could not load tenant users.' });
      }
    }

    // All other GET actions require a valid session token
    const session = validateSession(token);
    if (!session) return jsonOut({ success: false, message: 'Session expired. Please log in again.', sessionExpired: true, _debug: LAST_VALIDATE_DEBUG });
    applyTenantContext(session);

    if (action === 'getNextChallanNo')  return handleGetNextChallanNo();
    if (action === 'getChallanNos')     return handleGetChallanNos();
    if (action === 'getChallansData')   return handleGetChallansData();
    if (action === 'getInvoiceNos')     return handleGetInvoiceNos();
    if (action === 'getInvoiceByNo')    return handleGetInvoiceByNo(p.invoiceNo || '');
    if (action === 'getDashboard')      return handleGetDashboard();
    if (action === 'searchBuyer')       return handleSearchBuyer(p.q || '');
    if (action === 'getBuyers')         return handleGetBuyers();
    if (action === 'getItems')          return handleGetItems();
    if (action === 'getNextQuotationNo') return handleGetNextQuotationNo();
    if (action === 'getQuotationsData')  return handleGetQuotationsData();
    if (action === 'getQuotationByNo')   return handleGetQuotationByNo(p.quotNo || '');
    if (action === 'getNextCreditDebitNoteNo') return handleGetNextCreditDebitNoteNo(p.noteType || 'CREDIT');
    if (action === 'getCreditDebitNotesData')  return handleGetCreditDebitNotesData();
    if (action === 'getCreditDebitNoteByNo')   return handleGetCreditDebitNoteByNo(p.noteNo || '');
    if (action === 'getAMCData')         return handleGetAMCData();
    if (action === 'getNextCertNo')        return handleGetNextCertNo();
    if (action === 'getCertificates')      return handleGetCertificates();
    if (action === 'getCompanyInfo')       return handleGetCompanyInfo();

    return jsonOut({ success: false, message: 'Unknown action parameter' });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// ── POST ROUTER ──────────────────────────────────────────────────────────────

function doPost(e) {
  try {
    const body   = JSON.parse(e.postData.contents || '{}');
    const action = body.action || '';
    const token  = body.token  || '';

    // Login happens BEFORE any session exists, same as the super-admin
    // actions below. Moved here (POST, credentials in the body) instead of
    // doGet (was ?action=checkLogin&username=...&password=...) — a password
    // must never travel in a URL, since query strings can end up logged by
    // proxies/CDNs, cached, or leaked via the Referer header. Same
    // validateLoginSecret()/handleLogin() as before; only the transport
    // changed, so the login UI/behavior is unaffected.
    if (action === 'checkLogin') {
      if (!validateLoginSecret(body.secret || '')) return jsonOut({ success: false, message: 'Unauthorized Request' });
      // Own try/catch, not the outer one below: this path is reachable by
      // anyone with zero credentials, so an unexpected internal error here
      // must not echo raw exception text (sheet/internal details) back to
      // an unauthenticated caller the way the outer catch-all does for
      // already-authenticated actions.
      try {
        return handleLogin(body.username || '', body.password || '');
      } catch (loginErr) {
        Logger.log('checkLogin error: ' + loginErr);
        return jsonOut({ success: false, message: 'Login failed. Please try again.' });
      }
    }

    // Logout: deletes the session row outright so a captured/leaked token
    // can actually be revoked, instead of only being forgotten client-side
    // (sessionStorage.clear()) while remaining valid server-side until its
    // sliding-expiration TTL runs out.
    if (action === 'logout') {
      deleteSessionByToken(token);
      return jsonOut({ success: true });
    }

    // Super-admin-portal actions happen BEFORE any tenant session exists
    // (the portal manages tenants from outside any tenant's own login) —
    // gated by a separate super-admin secret (Script Properties →
    // SUPER_ADMIN_SECRET) instead of a session token.
    const SUPER_ADMIN_POST_ACTIONS = [
      'provisionTenant', 'updateTenant', 'deleteTenant',
      'addTenantUser', 'updateTenantUser', 'deleteTenantUser'
    ];
    if (SUPER_ADMIN_POST_ACTIONS.indexOf(action) > -1) {
      if (!isSuperAdminSecretValid(body.secret || '')) {
        return jsonOut({ success: false, message: 'Unauthorized: invalid super-admin secret.' });
      }
      if (action === 'provisionTenant') {
        try {
          const result = provisionTenant(body.data || {});
          return jsonOut({ success: true, tenantId: result.tenantId, spreadsheetId: result.spreadsheetId });
        } catch (provErr) {
          return jsonOut({ success: false, message: provErr.toString() });
        }
      }
      if (action === 'updateTenant') return handleUpdateTenant(body.tenantId || '', body.data || {});
      if (action === 'deleteTenant') return handleDeleteTenant(body.tenantId || '');
      if (action === 'addTenantUser') return handleAddTenantUser(body.tenantId || '', body.data || {});
      if (action === 'updateTenantUser') return handleUpdateTenantUser(body.username || '', body.data || {});
      if (action === 'deleteTenantUser') return handleDeleteTenantUser(body.username || '');
    }

    const session = validateSession(token);
    if (!session) return jsonOut({ success: false, message: 'Session expired. Please log in again.', sessionExpired: true, _debug: LAST_VALIDATE_DEBUG });
    applyTenantContext(session);

    // Restricted to the tenant's own Admin-role users — everything else a
    // logged-in user can do is routine day-to-day billing work that any
    // staff account needs, but changing company-wide branding/settings or
    // permanently deleting a contract are treated as admin-level actions.
    // A user's role defaults to 'Admin' when unset (see handleLogin), so
    // this only actually restricts anyone deliberately given a non-Admin
    // role going forward — no existing all-Admin account is affected.
    const ADMIN_ONLY_ACTIONS = ['saveCompanySettings', 'deleteAMCContract'];
    if (ADMIN_ONLY_ACTIONS.indexOf(action) > -1 && (session.role || 'Admin') !== 'Admin') {
      return jsonOut({ success: false, message: 'This action requires an Admin role.' });
    }

    if (action === 'addInvoice')        return handleAddInvoice(body.data, session.username);
    if (action === 'updateInvoice')     return handleUpdateInvoice(body.data, body.originalInvoiceNo || '', session.username);
    if (action === 'updateChallan')     return handleUpdateChallan(body, session.username);
    if (action === 'addBuyer')          return handleAddBuyer(body.data, session.username);
    if (action === 'addItem')           return handleAddItem(body.data, session.username);
    if (action === 'saveBuyer')         return handleSaveBuyer(body, session.username);
    if (action === 'saveChallan')       return handleSaveChallan(body, session.username);

// ── AUTH HANDLER ─────────────────────────────────────────────────────────────

function handleLogin(username, password) {
  username = (username || '').trim().toLowerCase();
  if (!username || !password) {
    return jsonOut({ success: false, message: 'Username and password fields are mandatory.' });
  }

  const usersSheet = getSheet(SHEET_USERS);
  const uData      = usersSheet.getDataRange().getValues();
  const uHeaders   = uData[0];
  let userRow = null, userRowIdx = -1;
  for (let i = 1; i < uData.length; i++) {
    if ((uData[i][0] || '').toString().trim().toLowerCase() === username) { userRow = uData[i]; userRowIdx = i; break; }
  }
  if (!userRow) return jsonOut({ success: false, message: 'Invalid credentials provided.' });

  const u = {};
  uHeaders.forEach(function(h, j) { u[h] = userRow[j]; });

  if ((u['Status'] || 'Active').toString().toLowerCase() === 'suspended') {
    return jsonOut({ success: false, message: 'This account has been suspended. Contact support.' });
  }

  if (!DEV_SKIP_PASSWORD_CHECK) {
    const matchedVersion = verifyPassword(password, u['Salt'] || '', u['Password Hash'] || '');
    if (!matchedVersion) {
      return jsonOut({ success: false, message: 'Invalid credentials provided.' });
    }
    // Transparent upgrade: a correct v1 (single-round) password gets
    // silently re-hashed with the stronger v2 algorithm and saved back —
    // no forced reset, nothing different for the user this login or next.
    if (matchedVersion === 'v1') {
      try {
        const hashCol = uHeaders.indexOf('Password Hash');
        if (hashCol > -1) {
          usersSheet.getRange(userRowIdx + 1, hashCol + 1).setValue(hashPasswordStrong(password, u['Salt'] || ''));
        }
      } catch (upgradeErr) {
        Logger.log('Password hash upgrade failed for ' + username + ': ' + upgradeErr);
      }
    }
  }

  const tenantId    = (u['Tenant ID'] || '').toString().trim();
  const tenantsSheet = getSheet(SHEET_TENANTS);
  const tData        = tenantsSheet.getDataRange().getValues();
  const tHeaders      = tData[0];
  let tenantRow = null;
  for (let i = 1; i < tData.length; i++) {
    if ((tData[i][0] || '').toString().trim() === tenantId) { tenantRow = tData[i]; break; }
  }
  if (!tenantRow) return jsonOut({ success: false, message: 'Tenant account not found. Contact support.' });

  const t = {};
  tHeaders.forEach(function(h, j) { t[h] = tenantRow[j]; });

  if ((t['Status'] || 'Active').toString().toLowerCase() !== 'active') {
    return jsonOut({ success: false, message: 'This account is currently inactive. Contact support.' });
  }

  const company = {
    name: t['Company Name'] || '', address: t['Address'] || '', gstin: t['GSTIN'] || '',
    state: t['State'] || 'Maharashtra', stateCode: t['State Code'] || '27',
    phone: t['Phone'] || '', email: t['Email'] || '', website: t['Website'] || '',
    bank: t['Bank'] || '', acctName: t['Acct Name'] || '', acctNo: t['Acct No'] || '',
    ifsc: t['IFSC'] || '', upi: t['UPI'] || '',
    pan: t['PAN'] || '', logo: t['Logo'] || '', signature: t['Signature'] || '',
    calendarId: t['Google Calendar ID'] || '', stamp: t['Stamp'] || '',
    watermark: t['Watermark'] || '', watermarkSrc: t['Watermark Image'] || '',
    watermarkStyle: t['Watermark Style'] || 'tilted'
  };

  const token = generateToken();
  const sessionData = {
    username:      username,
    displayName:   u['Display Name'] || username.toUpperCase(),
    tenantId:      tenantId,
    spreadsheetId: t['Spreadsheet ID'] || '',
    role:          u['Role'] || 'Admin',
    company:       company
  };
  storeSession(token, sessionData);
  return jsonOut({
    success:     true,
    token:       token,
    username:    username,
    displayName: sessionData.displayName,
    tenantId:    tenantId,
    tenantName:  company.name,
    company:     company,
    _debug: {
      tokenLength: token.length,
      storedInSheet: true
    }
  });
}

// Rewrites the Session Data JSON for one token so a settings change (e.g.
// Google Calendar ID) takes effect on the very next request instead of
// requiring the user to log out and back in. applyTenantContext() sets the
// in-memory COMPANY straight from this cached session snapshot on every
// request — without this, that snapshot would only ever refresh at login.
function updateSessionCompany(token, company) {
  if (!token) return;
  const sheet = getSheet(SHEET_SESSIONS);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === token) {
      try {
        const sessionData = JSON.parse(data[i][1]);
        sessionData.company = company;
        sheet.getRange(i + 1, 2).setValue(JSON.stringify(sessionData));
      } catch (e) {
        Logger.log('updateSessionCompany failed: ' + e);
      }
      return;
    }
  }
}

// A Google Sheets cell caps out at 50,000 characters — a resized-but-still-
// lossless PNG (needed for transparency) can blow past that in base64 for
// anything more detailed than a tiny icon. So a branding image is never
// stored inline: if the incoming value is a fresh data: URI (the browser
// just compressed/chroma-keyed an upload), it's saved to Drive and swapped
// for a short thumbnail-proxy link instead; if it's already a link
// (untouched from a previous save, or blank), it passes through as-is.
function maybeUploadBrandingImage(value, label) {
  if (!value || value.indexOf('data:') !== 0) return value || '';
  const match = value.match(/^data:([^;]+);base64,(.*)$/);
  if (!match) return value; // not a recognizable data URI — pass through defensively
  const mimeType = match[1];
  const ext = (mimeType.split('/')[1] || 'png').replace('jpeg', 'jpg');
  const fileName = label + ' - ' + new Date().getTime() + '.' + ext;
  const blob = Utilities.newBlob(Utilities.base64Decode(match[2]), mimeType, fileName);
  const folder = getOrCreateAppFolder('GME InvoiceApp - Branding');
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1000';
}

// Updates the calling tenant's own row in the (master-only) Tenants sheet —
// used by the Settings page's Branding card. Only touches the fields it's
// given; every other tenant column (name, GSTIN, bank details, etc.) is
// managed outside the app and is left untouched here.
function handleSaveCompanySettings(data, session, token) {
  data = data || {};
  const tenantId = (session && session.tenantId) || '';
  if (!tenantId) return jsonOut({ success: false, message: 'No tenant on this session.' });

  const sheet = getSheet(SHEET_TENANTS);
  const values = sheet.getDataRange().getValues();
  const headers = values[0];
  const idCol = headers.indexOf('Tenant ID');
  let rowIdx = -1;
  for (let i = 1; i < values.length; i++) {
    if ((values[i][idCol] || '').toString().trim() === tenantId) { rowIdx = i; break; }
  }
  if (rowIdx === -1) return jsonOut({ success: false, message: 'Tenant not found.' });

  const setCell = function(headerName, value) {
    const col = headers.indexOf(headerName);
    if (col === -1) return; // ensureHeaders() already ran via getSheet() above
    sheet.getRange(rowIdx + 1, col + 1).setValue(value);
  };

  const logoUrl = maybeUploadBrandingImage(data.logo || '', 'Logo');
  const signatureUrl = maybeUploadBrandingImage(data.signature || '', 'Signature');
  const stampUrl = maybeUploadBrandingImage(data.stamp || '', 'Stamp');
  const watermarkUrl = maybeUploadBrandingImage(data.watermark || '', 'Watermark');
  // watermarkSrc is the tenant's raw uploaded watermark photo (separate
  // from Logo) — kept as its own Drive-hosted image so the client can
  // recompute a fresh Tilted/Straight version later without re-uploading.
  // 'Watermark' (watermarkUrl above) stays the ALREADY-processed (rotated +
  // faded) image every print engine actually renders — nothing downstream
  // needs to know this field exists at all.
  const watermarkSrcUrl = maybeUploadBrandingImage(data.watermarkSrc || '', 'Watermark Image');
  const watermarkStyle = (data.watermarkStyle === 'straight') ? 'straight' : 'tilted';

  setCell('PAN', data.pan || '');
  setCell('Logo', logoUrl);
  setCell('Signature', signatureUrl);
  setCell('Google Calendar ID', data.calendarId || '');
  setCell('Stamp', stampUrl);
  setCell('Watermark', watermarkUrl);
  setCell('Watermark Image', watermarkSrcUrl);
  setCell('Watermark Style', watermarkStyle);

  // Refresh both the in-memory COMPANY (for the rest of this request) and
  // the session's cached snapshot (for every request after it) so the new
  // Calendar ID / branding images are live immediately — no logout/login needed.
  COMPANY = Object.assign({}, COMPANY, {
    pan: data.pan || '', logo: logoUrl, signature: signatureUrl,
    calendarId: data.calendarId || '', stamp: stampUrl, watermark: watermarkUrl,
    watermarkSrc: watermarkSrcUrl, watermarkStyle: watermarkStyle
  });
  updateSessionCompany(token, COMPANY);

  return jsonOut({ success: true, company: COMPANY });
}

// ── TENANT PROVISIONING ──────────────────────────────────────────────────────
// Called via doPost action=provisionTenant (super-admin-secret gated), or
// directly from the Apps Script editor. Creates a brand-new, fully isolated
// Google Sheet for the tenant's business data and registers the tenant +
// their first admin login in the control spreadsheet's Tenants/Users sheets.
function provisionTenant(input) {
  input = input || {};
  if (!input.companyName)    throw new Error('companyName is required.');
  if (!input.adminUsername)  throw new Error('adminUsername is required.');
  if (!input.adminPassword)  throw new Error('adminPassword is required.');

  const tenantsSheetChk = getSheet(SHEET_TENANTS);
  const existingTenants = tenantsSheetChk.getDataRange().getValues();
  var tenantId = (input.tenantId && input.tenantId.trim())
    ? input.tenantId.trim()
    : 'T-' + Utilities.getUuid().replace(/-/g, '').substring(0, 10).toUpperCase();
  const dupeTenant = existingTenants.slice(1).some(function(r) {
    return (r[0] || '').toString().trim() === tenantId;
  });
  if (dupeTenant) throw new Error('Tenant ID "' + tenantId + '" is already in use.');

  const adminUsername = input.adminUsername.trim().toLowerCase();
  const usersSheetChk  = getSheet(SHEET_USERS);
  const existingUsers  = usersSheetChk.getDataRange().getValues();
  const dupeUser = existingUsers.slice(1).some(function(r) {
    return (r[0] || '').toString().trim().toLowerCase() === adminUsername;
  });
  if (dupeUser) throw new Error('Username "' + adminUsername + '" is already taken.');

  // 1. Create a dedicated, fully isolated spreadsheet for this tenant's data.
  const newSs = SpreadsheetApp.create(input.companyName + ' — GME InvoiceApp Data');
  const newSpreadsheetId = newSs.getId();
  try {
    const defaultSheet = newSs.getSheetByName('Sheet1');
    if (defaultSheet) newSs.deleteSheet(defaultSheet);
  } catch (e) { /* non-fatal */ }

  // 2. Register the tenant + their company/bank details (used on every
  //    printed document and emailed PDF for this tenant). Header-mapped
  //    (not a fixed positional array) so this keeps working correctly as
  //    columns get added to TENANT_HEADERS over time — Logo/Signature/Stamp
  //    are deliberately left blank here; the tenant uploads those
  //    themselves from their own Settings page after their first login.
  const tenantsSheet = getSheet(SHEET_TENANTS);
  const tHeaders = tenantsSheet.getRange(1, 1, 1, tenantsSheet.getLastColumn()).getValues()[0]
    .map(function(h) { return String(h).trim(); });
  const tenantFields = {
    'Tenant ID': tenantId, 'Company Name': input.companyName || '', 'Address': input.address || '',
    'GSTIN': input.gstin || '', 'State': input.state || 'Maharashtra', 'State Code': input.stateCode || '27',
    'Phone': input.phone || '', 'Email': input.email || '', 'Website': input.website || '',
    'Bank': input.bank || '', 'Acct Name': input.acctName || '', 'Acct No': input.acctNo || '',
    'IFSC': input.ifsc || '', 'UPI': input.upi || '', 'Spreadsheet ID': newSpreadsheetId,
    'Status': 'Active', 'Plan': input.plan || 'Trial',
    'Created Date': new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' }),
    'PAN': input.pan || '', 'Google Calendar ID': input.calendarId || ''
  };
  tenantsSheet.appendRow(tHeaders.map(function(h) { return tenantFields.hasOwnProperty(h) ? tenantFields[h] : ''; }));

  // 3. Create their first admin login.
  const salt = generateSalt();
  const hash = hashPasswordStrong(input.adminPassword, salt);
  const usersSheet = getSheet(SHEET_USERS);
  usersSheet.appendRow([
    adminUsername, hash, salt, input.adminDisplayName || input.adminUsername,
    tenantId, 'Admin', 'Active',
    new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' })
  ]);

  return { tenantId: tenantId, spreadsheetId: newSpreadsheetId };
}

// ── SUPER ADMIN: LIST / UPDATE / DELETE TENANTS ─────────────────────────────
// All three are gated by isSuperAdminSecretValid() at the router, never by a
// login token — this portal manages tenants from outside any tenant's login.

function handleListTenants() {
  try {
    const sheet = getSheet(SHEET_TENANTS);
    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(function(h) { return String(h).trim(); });

    const usersSheet = getSheet(SHEET_USERS);
    const usersData = usersSheet.getDataRange().getValues();
    const userHeaders = usersData[0].map(function(h) { return String(h).trim(); });
    const tidCol = userHeaders.indexOf('Tenant ID');
    const userCounts = {};
    for (var i = 1; i < usersData.length; i++) {
      var tid = (tidCol > -1 ? usersData[i][tidCol] : '').toString().trim();
      if (tid) userCounts[tid] = (userCounts[tid] || 0) + 1;
    }

    const tenants = [];
    for (var r = 1; r < data.length; r++) {
      var row = {};
      headers.forEach(function(h, j) { row[h] = data[r][j]; });
      row['User Count'] = userCounts[String(row['Tenant ID']).trim()] || 0;
      // Logo/Signature/Stamp can be sizeable base64 strings — never needed
      // for the tenant list/edit screens, so drop them from the payload
      // rather than shipping them over the wire for no reason.
      delete row['Logo']; delete row['Signature']; delete row['Stamp'];
      tenants.push(row);
    }
    return jsonOut({ success: true, tenants: tenants });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// Removes the tenant's own row and revokes every one of its users' logins
// (plus any currently-active sessions, so access ends immediately rather
// than waiting out the session TTL) — but deliberately does NOT touch
// their dedicated Google Sheet (their actual invoices/business data),
// which stays intact in Drive in case this needs to be reversed or the
// data re-attached to a new tenant later. The response returns that
// Spreadsheet ID so the super admin knows exactly where the data still lives.
function handleDeleteTenant(tenantId) {
  try {
    tenantId = (tenantId || '').trim();
    if (!tenantId) return jsonOut({ success: false, message: 'Tenant ID is required.' });

    const sheet = getSheet(SHEET_TENANTS);
    const values = sheet.getDataRange().getValues();
    const headers = values[0].map(function(h) { return String(h).trim(); });
    const idCol = headers.indexOf('Tenant ID');
    const ssCol = headers.indexOf('Spreadsheet ID');
    var rowIdx = -1, spreadsheetId = '';
    for (var i = 1; i < values.length; i++) {
      if ((values[i][idCol] || '').toString().trim() === tenantId) {
        rowIdx = i;
        spreadsheetId = ssCol > -1 ? values[i][ssCol] : '';
        break;
      }
    }
    if (rowIdx === -1) return jsonOut({ success: false, message: 'Tenant not found.' });

    sheet.deleteRow(rowIdx + 1);

    const usersSheet = getSheet(SHEET_USERS);
    const usersData = usersSheet.getDataRange().getValues();
    const userHeaders = usersData[0].map(function(h) { return String(h).trim(); });
    const utidCol = userHeaders.indexOf('Tenant ID');
    for (var u = usersData.length - 1; u >= 1; u--) {
      if ((usersData[u][utidCol] || '').toString().trim() === tenantId) {
        usersSheet.deleteRow(u + 1);
      }
    }

    try {
      const sessionsSheet = getSheet(SHEET_SESSIONS);
      const sessionsData = sessionsSheet.getDataRange().getValues();
      for (var s = sessionsData.length - 1; s >= 1; s--) {
        try {
          const sessionData = JSON.parse(sessionsData[s][1] || '{}');
          if (sessionData.tenantId === tenantId) sessionsSheet.deleteRow(s + 1);
        } catch (parseErr) { /* malformed row — skip */ }
      }
    } catch (sessErr) {
      Logger.log('Session cleanup during tenant delete failed: ' + sessErr);
    }

    return jsonOut({ success: true, spreadsheetId: spreadsheetId });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// Refuses to delete a tenant's LAST remaining login — that would silently
// lock the tenant out of their own account with no way back in short of
// the super admin adding a fresh user by hand. Deleting the whole tenant
// (handleDeleteTenant) is the deliberate way to remove all access at once.
function handleDeleteTenantUser(username) {
  try {
    username = (username || '').trim().toLowerCase();
    if (!username) return jsonOut({ success: false, message: 'Username is required.' });

    const sheet = getSheet(SHEET_USERS);
    const values = sheet.getDataRange().getValues();
    const headers = values[0].map(function(h) { return String(h).trim(); });
    const userCol = headers.indexOf('Username');
    const tidCol = headers.indexOf('Tenant ID');
    var rowIdx = -1, tenantId = '';
    for (var i = 1; i < values.length; i++) {
      if ((values[i][userCol] || '').toString().trim().toLowerCase() === username) {
        rowIdx = i; tenantId = (values[i][tidCol] || '').toString().trim();
        break;
      }
    }
    if (rowIdx === -1) return jsonOut({ success: false, message: 'User not found.' });

    const siblingCount = values.slice(1).filter(function(r) {
      return (r[tidCol] || '').toString().trim() === tenantId;
    }).length;
    if (siblingCount <= 1) {
      return jsonOut({ success: false, message: 'Cannot delete the only remaining login for this tenant — add another user first, or delete the whole tenant instead.' });
    }

    sheet.deleteRow(rowIdx + 1);

    try {
      const sessionsSheet = getSheet(SHEET_SESSIONS);
      const sessionsData = sessionsSheet.getDataRange().getValues();
      for (var s = sessionsData.length - 1; s >= 1; s--) {
        try {
          const sessionData = JSON.parse(sessionsData[s][1] || '{}');
          if ((sessionData.username || '').toString().trim().toLowerCase() === username) sessionsSheet.deleteRow(s + 1);
        } catch (parseErr) { /* malformed row — skip */ }
      }
    } catch (sessErr) {
      Logger.log('Session cleanup during user delete failed: ' + sessErr);
    }

    return jsonOut({ success: true });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// ── CHALLAN NUMBER AUTO-INCREMENT ────────────────────────────────────────────
//
// Format: CH/YY-YY/NNN  e.g.  CH/26-27/001
// Reads all existing challan numbers from the Challans sheet,
// finds the highest sequence number, and returns next one.
// Falls back to CH/FY/001 if sheet is empty.

function handleGetNextChallanNo() {
  try {
    const sheet = getSheet(SHEET_CHALLANS);
    const data  = sheet.getDataRange().getValues();

    // Column A (index 0) = Challan No
    const existingNos = data.slice(1)
      .map(r => (r[0] || '').toString().trim())
      .filter(v => v !== '');

    // Determine current financial year suffix e.g. "26-27"
    const now = new Date();
    const yr  = now.getFullYear();
    const mo  = now.getMonth() + 1; // 1-based
    const fyStart = mo >= 4 ? yr : yr - 1;
    const fyEnd   = fyStart + 1;
    const fySuffix = String(fyStart).slice(-2) + '-' + String(fyEnd).slice(-2); // "26-27"
    const prefix   = 'CH/' + fySuffix + '/';

    // Find the highest sequence number for this FY
    let maxSeq = 0;
    existingNos.forEach(no => {
      if (no.toUpperCase().startsWith(prefix.toUpperCase())) {
        const parts = no.split('/');
        const seq   = parseInt(parts[parts.length - 1], 10);
        if (!isNaN(seq) && seq > maxSeq) maxSeq = seq;
      }
    });

    const nextSeq = maxSeq + 1;
    const challanNo = prefix + String(nextSeq).padStart(3, '0');

    return jsonOut({ success: true, challanNo: challanNo });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// AUTO EMAIL — INVOICE PDF SENDER
// ═══════════════════════════════════════════════════════════════════════════
//
// Fires automatically after every successful invoice save.
// Builds a self-contained inline-CSS HTML invoice, converts it to PDF
// via Utilities.newBlob().getAs('application/pdf'), and sends it to the
// fixed SELLER_EMAIL address (itransform007@gmail.com) via MailApp.
//
// No buyer email is used — the PDF goes to the seller only.
//
// Called by handleAddInvoice() after sheet.appendRow() succeeds.
// Returns: { sent: true/false, message: string }
// ─────────────────────────────────────────────────────────────────────────

// ═══════════════════════════════════════════════════════════════════════════
// LEDGERPRINT (GAS PORT) — line-by-line port of the browser's LedgerPrint
// print engine (index.html), so the WhatsApp/email PDF is built by the EXACT
// same layout logic as the front-end Print/PDF button, instead of a
// hand-maintained parallel table-based reimplementation that kept drifting
// out of sync (border widths, missing rate labels, a duplicated "Only" —
// every one of those was the two versions quietly disagreeing). This IS the
// front-end's document model; only the final "turn it into a PDF" step
// differs, since GAS has no browser to print from — it goes through Drive's
// HTML-to-PDF conversion instead (the same temp-Drive-file trick already
// used elsewhere in this file).
//
// ONE deliberate deviation from a literal port: the browser's stamp+signature
// overlay uses position:absolute + transform to center them on each other.
// That specific technique broke Drive's PDF conversion (a stray vertical
// line down the whole page) when it was tried directly server-side — see
// bankBlock() below, which uses the negative-margin technique instead: the
// same fix already proven to work in the old Gscript-native renderer.
// Nothing else is changed from the browser version.
//
// The OLD hand-rolled renderer (buildInvoiceHtml and everything it calls —
// computeInvoiceLineCalc, sellerHtml, partiesHtml, bankBandHtml,
// sigStampCellHtml, totalsHtml, itemsTableHtml, INV_GEO, INV_AFM, etc.) is
// left in place below, just unused now — not deleted. There's no version
// control on this file, and this is a genuinely new, untested-in-Drive code
// path in a live invoicing app; keeping the old renderer intact means
// reverting the three call sites this replaces is a small, quick edit if
// Drive's conversion doesn't handle flexbox as well as hoped. Once this is
// confirmed working end-to-end, the old renderer can be deleted.
// ═══════════════════════════════════════════════════════════════════════════
var LedgerPrintGas = (function () {

  var PT_TO_MM = 0.352778, DEFAULT_ADVANCE = 556;

  function textWidthMM(text, fontPt) {
    var s = String(text == null ? '' : text), u = 0;
    for (var i = 0; i < s.length; i++) {
      var c = s.charCodeAt(i);
      u += (AFM[c] === undefined ? DEFAULT_ADVANCE : AFM[c]);
    }
    return u / 1000 * fontPt * PT_TO_MM;
  }

  function wrappedLineCount(text, widthMM, fontPt) {
    var words = String(text == null ? '' : text).trim().split(/\s+/).filter(Boolean);
    if (!words.length) return 1;
    var space = textWidthMM(' ', fontPt), lines = 1, used = 0;
    for (var i = 0; i < words.length; i++) {
      var w = textWidthMM(words[i], fontPt);
      if (w > widthMM) {
        if (used > 0) { lines++; used = 0; }
        var rows = Math.ceil(w / widthMM);
        lines += rows - 1;
        used = w - (rows - 1) * widthMM;
        continue;
      }
      var add = used === 0 ? w : space + w;
      if (used + add > widthMM) { lines++; used = w; } else { used += add; }
    }
    return lines;
  }

  function esc(v) {
    return String(v == null ? '' : v).replace(/[&<>"']/g, function (c) {
      return { '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c];
    });
  }
  function num(v) {
    var n = parseFloat(String(v == null ? '' : v).replace(/,/g, ''));
    return isFinite(n) ? n : 0;
  }
  function r2(n) { return Math.round((n + Number.EPSILON) * 100) / 100; }
  var nfIN = new Intl.NumberFormat('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  function fmtNum(n) { return nfIN.format(r2(n)); }
  function fmtCur(n) { return '₹' + nfIN.format(r2(n)); }

  var MON = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  function dmy(v) {
    if (!v) return '';
    var d = new Date(v);
    if (isNaN(d.getTime())) return String(v);
    return ('0' + d.getDate()).slice(-2) + '-' + MON[d.getMonth()] + '-' + d.getFullYear();
  }
  function fyOf(v) {
    var d = v ? new Date(v) : new Date();
    if (isNaN(d.getTime())) d = new Date();
    var y = d.getFullYear(), start = d.getMonth() >= 3 ? y : y - 1;
    return start + '-' + String(start + 1).slice(2);
  }

  var W_ONES = ['','One','Two','Three','Four','Five','Six','Seven','Eight','Nine','Ten',
    'Eleven','Twelve','Thirteen','Fourteen','Fifteen','Sixteen','Seventeen','Eighteen','Nineteen'];
  var W_TENS = ['','','Twenty','Thirty','Forty','Fifty','Sixty','Seventy','Eighty','Ninety'];
  function two(n) {
    if (n < 20) return W_ONES[n];
    var t = Math.floor(n / 10), o = n % 10;
    return W_TENS[t] + (o ? ' ' + W_ONES[o] : '');
  }
  function intWords(n) {
    if (n === 0) return 'Zero';
    var p = [];
    var crore = Math.floor(n / 1e7); n %= 1e7;
    var lakh = Math.floor(n / 1e5); n %= 1e5;
    var thousand = Math.floor(n / 1e3); n %= 1e3;
    var hundred = Math.floor(n / 100), rest = n % 100;
    if (crore) p.push(intWords(crore) + ' Crore');
    if (lakh) p.push(two(lakh) + ' Lakh');
    if (thousand) p.push(two(thousand) + ' Thousand');
    if (hundred) p.push(W_ONES[hundred] + ' Hundred');
    if (rest) p.push(two(rest));
    return p.join(' ');
  }
  function numberToIndianWords(amount) {
    var neg = amount < 0, abs = Math.abs(r2(amount));
    var rupees = Math.floor(abs + 1e-9), paise = Math.round((abs - rupees) * 100);
    var s = 'Rupees ' + intWords(rupees);
    if (paise) s += ' and ' + two(paise) + ' Paise';
    return (neg ? 'Minus ' : '') + s + ' Only';
  }

  function buildDoc(kind, data, company) {
    DOC = DOCS[kind] || DOCS.invoice;
    company = company || {};
    var cgstRate = num(data.cgstRate), sgstRate = num(data.sgstRate), igstRate = num(data.igstRate);
    var subTotal = num(data.subTotal);
    var cgst = data.cgstAmount != null ? num(data.cgstAmount) : r2(subTotal * cgstRate / 100);
    var sgst = data.sgstAmount != null ? num(data.sgstAmount) : r2(subTotal * sgstRate / 100);
    var igst = data.igstAmount != null ? num(data.igstAmount) : r2(subTotal * igstRate / 100);

    var intra = (cgst > 0 || sgst > 0) || (igst === 0 && cgstRate + sgstRate > 0);
    var invoiceRate = intra ? r2(cgstRate + sgstRate) : igstRate;

    if (kind === 'quotation' && cgst === 0 && sgst === 0 && igst === 0 && num(data.taxTotal) > 0) {
      var qIntra = !(company.stateCode && data.stateCode) || String(company.stateCode) === String(data.stateCode);
      var qTax = r2(num(data.taxTotal));
      if (qIntra) { cgst = r2(qTax / 2); sgst = r2(qTax - cgst); }
      else { igst = qTax; }
      intra = qIntra;
      invoiceRate = 0;
    }

    // A genuinely tax-exempt invoice (GST Type = No GST) is a materially
    // different document from a "Tax Invoice" — swap the header title
    // rather than print one titled "Tax Invoice" that charges no tax.
    // DOC is reassigned to a fresh copy here (not mutated in place), so
    // this can never leak into DOCS.invoice itself for a later document
    // built within the same execution.
    if (kind === 'invoice' && cgst === 0 && sgst === 0 && igst === 0) {
      DOC = Object.assign({}, DOC, { title: 'Non-tax invoice' });
    }

    var raw = data.items || [];
    var lines = raw.map(function (it, i) {
      var qty = num(it.qty), rate = num(it.rate);
      var taxable = it.amount != null ? num(it.amount) : r2(qty * rate);
      var gstPct = it.gstPct != null ? num(it.gstPct) : invoiceRate;
      return {
        sn: i + 1,
        description: it.desc || it.description || '',
        hsn: it.hsn || '',
        unit: it.unit || 'Nos',
        qty: qty, rate: rate, taxable: taxable, gstRate: gstPct,
        remarks: it.remarks || ''
      };
    });

    var linesTotal = r2(lines.reduce(function (a, l) { return a + l.taxable; }, 0));

    if (DOC.money && lines.length && subTotal > 0 && Math.abs(linesTotal - subTotal) > 0.05) {
      throw new Error('Line items total ' + linesTotal.toFixed(2) +
        ' but the document states ' + subTotal.toFixed(2) +
        '. Re-save the document before printing.');
    }
    var totalTax = r2(cgst + sgst + igst);
    var grandTotal = data.grandTotal != null ? num(data.grandTotal) : Math.round(subTotal + totalTax);
    var roundOff = r2(grandTotal - (subTotal + totalTax));

    return {
      seller: {
        name: company.name || '', address: company.address || '',
        gstin: company.gstin || '', state: company.state || '',
        stateCode: company.stateCode || '', phone: company.phone || '',
        email: company.email || '', website: company.website || '',
        pan: company.pan || '', logo: company.logo || '',
        signature: company.signature || '', stamp: company.stamp || '',
        watermark: company.watermark || ''
      },
      bank: {
        accountName: company.acctName || company.name || '',
        accountNumber: company.acctNo || '', ifsc: company.ifsc || '',
        bankName: company.bank || '', upi: company.upi || ''
      },
      customer: {
        name: data.buyerName || '', address: data.buyerAddress || '',
        gstin: data.buyerGstin || '', state: data.buyerState || '',
        stateCode: data.stateCode || '', phone: data.buyerPhone || '',
        email: data.buyerEmail || ''
      },
      kind: kind,
      doc: DOC,
      meta: {
        invoiceNo: data.invoiceNo || data.quotNo || data.noteNo || data.challanNo || '',
        date: data.date || '',
        challanNo: data.challanNo || '', challanDate: data.challanDate || '',
        refNo: data.refNo || data.refInvoiceNo || data.againstInvoiceNo || '',
        refDate: data.againstInvoiceDate || '',
        validUntil: data.validUntil || '',
        transport: data.transport || '',
        reason: data.reason || '',
        notes: data.notes || '',
        totalQty: r2((data.items || []).reduce(function (a, it) { return a + num(it.qty); }, 0)),
        placeOfSupply: data.placeOfSupply || data.buyerState || '',
        placeOfSupplyCode: data.stateCode || '',
        paymentTerms: data.paymentTerms || '', dueDate: data.dueDate || '',
        reverseCharge: !!data.reverseCharge
      },
      lines: lines,
      intra: intra, invoiceRate: invoiceRate,
      cgstRate: cgstRate, sgstRate: sgstRate, igstRate: igstRate,
      subTotal: subTotal, linesTotal: linesTotal,
      cgst: cgst, sgst: sgst, igst: igst, totalTax: totalTax,
      roundOff: roundOff, grandTotal: grandTotal,
      words: numberToIndianWords(grandTotal),
      groups: DOC.tax ? taxGroups(lines, intra, invoiceRate) : { rows: [], intra: intra }
    };
  }
  function buildInvoice(data, company) { return buildDoc('invoice', data, company); }

  function taxGroups(lines, intra, invoiceRate) {
    function build(keyFn) {
      var map = {}, order = [];
      lines.forEach(function (l) {
        var k = keyFn(l);
        if (!map[k]) { map[k] = { hsn: l.hsn || '-', gstRate: l.gstRate, taxable: 0 }; order.push(k); }
        map[k].taxable = r2(map[k].taxable + l.taxable);
      });
      return order.map(function (k) { return map[k]; });
    }
    var g = build(function (l) { return l.hsn + '|' + l.gstRate; });
    if (g.length > 8) {
      g = build(function (l) { return String(l.gstRate); });
      g.forEach(function (x) { x.hsn = 'Various'; });
    }
    g.forEach(function (x) {
      var rate = x.gstRate == null ? invoiceRate : x.gstRate;
      x.cgst = intra ? r2(x.taxable * rate / 200) : 0;
      x.sgst = intra ? r2(x.taxable * rate / 200) : 0;
      x.igst = intra ? 0 : r2(x.taxable * rate / 100);
      x.tax = r2(x.cgst + x.sgst + x.igst);
    });
    g.sort(function (a, b) { return a.gstRate - b.gstRate || String(a.hsn).localeCompare(String(b.hsn)); });
    return { rows: g, intra: intra };
  }

  function headerHeight(first) {
    return first ? GEO.titleH + GEO.sellerH + GEO.partiesH + GEO.stripH
                 : GEO.titleH + GEO.contStripH;
  }
  function taxBlockHeight(g) { return GEO.taxHeadH + g * GEO.taxRowH + GEO.taxFootH; }
  function footerHeight(g) {
    return (DOC.tax ? taxBlockHeight(g) : 0) +
      GEO.totalsH + GEO.wordsH + GEO.bankH + GEO.pageFootH;
  }
  function tableAreaHeight(first, g) {
    return r2(GEO.pageH - headerHeight(first) - footerHeight(g));

  function bodyHeight(first, g) {
    return r2(GEO.pageH - headerHeight(first) - footerHeight(g) - GEO.tableHeadH);
  }
  function calculatePageCapacity(first, g) {
    return Math.max(1, Math.floor(bodyHeight(first, g) / GEO.rowH));
  }

  function paginateItems(lines, g) {
    var units = lines.map(rowUnitsFor), pages = [], i = 0;
    while (i < lines.length) {
      var first = pages.length === 0;
      var cap = calculatePageCapacity(first, g);
      var rest = 0;
      for (var k = i; k < lines.length; k++) rest += units[k];
      if (rest <= cap) {
        pages.push({ from: i, items: lines.slice(i), first: first, last: true, used: rest });
        i = lines.length;
      } else {
        var used = 0, j = i;
        while (j < lines.length && used + units[j] <= cap) { used += units[j]; j++; }
        if (j === i) { used = units[i]; j = i + 1; }
        pages.push({ from: i, items: lines.slice(i, j), first: first, last: false, used: used });
        i = j;
      }
    }
    if (!pages.length) pages.push({ from: 0, items: [], first: true, last: true, used: 0 });
    pages.forEach(function (p, idx) {
      p.number = idx + 1;
      p.capacity = calculatePageCapacity(p.first, g);
      p.tableH = tableAreaHeight(p.first, g);
      p.bodyH = r2(p.tableH - GEO.tableHeadH);
      p.filler = r2(p.bodyH - p.used * GEO.rowH);
    });
    return pages;
  }

  function itemRow(l) {

  function partyLines(p) {
    var out = [];
    String(p.address || '').split(/\r?\n/).forEach(function (t) { if (t.trim()) out.push(t.trim()); });
    if (p.gstin) out.push('GSTIN: ' + p.gstin);
    if (p.state) out.push('State: ' + p.state + (p.stateCode ? ' (code ' + p.stateCode + ')' : ''));
    if (p.phone) out.push('Phone: ' + p.phone);
    return out.map(esc).join('<br>');
  }

  function metaRows(inv) {
    var m = inv.meta, k = inv.kind, out = [];
    function row(label, v) {
      return '<tr style="height:' + GEO.metaRowH + 'mm"><td>' + esc(label) + '</td>' +
        '<td class="bold">' + esc(v || '-') + '</td></tr>';
    }
    var noun = { invoice:'Invoice', quotation:'Quotation', creditnote:'Credit note',
                 debitnote:'Debit note', challan:'Challan' }[k] || 'Document';
    out.push(row(noun + ' no.', m.invoiceNo));
    out.push(row(noun + ' date', dmy(m.date)));
    if (k === 'invoice') {
      out.push(row('Challan no.', m.challanNo));
      if (m.challanDate) out.push(row('Challan date', dmy(m.challanDate)));
    }
    if (k === 'quotation') {
      out.push(row('Reference', m.refNo));
      out.push(row('Valid until', m.validUntil ? dmy(m.validUntil) : ''));
    }
    if (k === 'creditnote' || k === 'debitnote') {
      out.push(row('Against invoice', m.refNo));
      out.push(row('Invoice date', m.refDate ? dmy(m.refDate) : ''));
    }
    if (k === 'challan') {
      out.push(row('Against invoice', m.refNo));
      out.push(row('Transport', m.transport));
    }
    var pos = m.placeOfSupply
      ? m.placeOfSupply + (m.placeOfSupplyCode ? ' (' + m.placeOfSupplyCode + ')' : '') : '';
    out.push(row('Place of supply', pos));
    if (k !== 'challan') out.push(row('Destination', inv.customer.state));
    if (m.dueDate) out.push(row('Due date', dmy(m.dueDate)));
    if (m.reverseCharge) out.push(row('Reverse charge', 'Yes, section 9(3)/9(4)'));
    return out.join('');
  }

    var showShip = inv.kind !== 'invoice';

    return '<div class="inv-parties" style="height:' + GEO.partiesH + 'mm">' +
      '<div class="bill"><span class="lbl">Bill to</span>' +
        '<div class="party-name">' + esc(inv.customer.name) + '</div>' +
        '<div class="party-lines">' + partyLines(inv.customer) + '</div></div>' +
      (showShip
        ? '<div class="ship"><span class="lbl">Ship to</span>' +
          '<div class="party-name">' + esc(sh.name || inv.customer.name) + '</div>' +
          '<div class="party-lines">' + partyLines(sh) + '</div></div>'
        : '') +
      '<div class="meta"><table>' + metaRows(inv) + '</table></div></div>';
  }

  function stripBlock(inv) {
    var bits = [];
    if (inv.doc.tax) {
      bits.push(inv.intra ? 'Intra-state supply, CGST and SGST applicable'
                          : 'Inter-state supply, IGST applicable');
    } else {
      bits.push('Delivery challan, not a tax invoice');
    }
    if (inv.kind === 'quotation' && inv.meta.validUntil)
      bits.push('Valid until: ' + dmy(inv.meta.validUntil));
    if (inv.meta.paymentTerms) bits.push('Payment terms: ' + inv.meta.paymentTerms);
    if (inv.meta.dueDate) bits.push('Due date: ' + dmy(inv.meta.dueDate));
    if (inv.doc.money) bits.push('Currency: INR');
    bits.push('FY ' + fyOf(inv.meta.date));
    return '<div class="inv-strip" style="height:' + GEO.stripH + 'mm">' +
      bits.map(function (b) { return '<span>' + esc(b) + '</span>'; }).join('') + '</div>';
  }

  function taxTable(inv) {
    var g = inv.groups.rows, intra = inv.intra;
    var head = intra
      ? '<th style="width:22%">HSN/SAC</th><th style="width:22%">Taxable value</th>' +
        '<th style="width:8%">CGST %</th><th style="width:16%">CGST amount</th>' +
        '<th style="width:8%">SGST %</th><th style="width:16%">SGST amount</th><th style="width:8%">Total tax</th>'
      : '<th style="width:26%">HSN/SAC</th><th style="width:28%">Taxable value</th>' +
        '<th style="width:10%">IGST %</th><th style="width:22%">IGST amount</th><th style="width:14%">Total tax</th>';
    var body = g.map(function (x) {
      return intra

  }

// Converts a LedgerPrintGas-built document into a PDF Blob via the existing
// temp-Drive-file trick (Utilities.newBlob().getAs() for PDF only works on a
// DriveApp File, not directly on a blob — see the ORIGINAL renderer's own
// comment on this same quirk, a few hundred lines below).
function renderLedgerPrintPdf(data, company, kind, fileName) {
  var built = LedgerPrintGas.buildDocument(data, company, kind || 'invoice');
  var tempFile = null, pdfBlob;
  try {
    tempFile = DriveApp.createFile(Utilities.newBlob(built.document, 'text/html', 'temp_ledgerprint.html'));
    pdfBlob = tempFile.getAs('application/pdf');
    pdfBlob.setName(fileName || ((built.invoice.meta.invoiceNo || 'Document') + '.pdf'));
  } finally {
    if (tempFile) { try { tempFile.setTrashed(true); } catch (e) { Logger.log('Temp file cleanup: ' + e); } }
  }
  return pdfBlob;
}

    '  Buyer Name  : ' + (data.buyerName || '-') + '\n' +
    '  Buyer GSTIN : ' + (data.buyerGstin || 'Unregistered') + '\n' +
    '  Grand Total : ' + grandTotalFormatted + '\n' +
    (data.challanNo ? '  Challan No. : ' + data.challanNo + '\n' : '') +
    '\nPlease find the invoice PDF attached.\n\n' +
    '-- Enterprises Invoice System';

  // ── Send email to seller ────────────────────────────────────────────────
  MailApp.sendEmail(SELLER_EMAIL, subject, emailBody, {
    name:        'Enterprises — Invoice System',
    attachments: [pdfBlob]
  });

  Logger.log('Invoice email sent to seller ' + SELLER_EMAIL + ' for ' + (data.invoiceNo || ''));

}

// Per-line GST — each item carries its own gstPct, and CGST/SGST vs IGST is
// decided by comparing seller/buyer state codes (or an explicit override),
// exactly like the browser-side print engine (renderInvoicePages() in the
// frontend). Mirroring that logic here is what keeps the emailed PDF
// consistent with what Print/Reprint show on screen.
function computeInvoiceLineCalc(data) {
  const items = (data.items || []).filter(function(it){ return it && it.desc && String(it.desc).trim() !== ''; });
  const mode = data.taxMode || 'auto';
  let intra;
  if (mode === 'intra') intra = true;
  else if (mode === 'inter') intra = false;
  else {
    const a = String(COMPANY.stateCode || ''), b = String(data.stateCode || '');
    intra = Boolean(a && b && a === b);
  }
  const lines = items.map(function(it) {
    const qty = parseFloat(it.qty || 0), rate = parseFloat(it.rate || 0);
    const taxable = Math.round(qty * rate * 100) / 100;
    const gstPct = parseFloat(it.gstPct != null ? it.gstPct : 18);
    const cgst = intra ? Math.round(taxable * gstPct / 200 * 100) / 100 : 0;
    const sgst = intra ? Math.round(taxable * gstPct / 200 * 100) / 100 : 0;
    const igst = intra ? 0 : Math.round(taxable * gstPct / 100 * 100) / 100;
    return { desc: it.desc, hsn: it.hsn, unit: it.unit, qty: qty, rate: rate, taxable: taxable,
             gstPct: gstPct, cgst: cgst, sgst: sgst, igst: igst };
  });
  const sum = function(k) { return Math.round(lines.reduce(function(a, l) { return a + l[k]; }, 0) * 100) / 100; };
  const taxable = sum('taxable'), cgst = sum('cgst'), sgst = sum('sgst'), igst = sum('igst');
  const totalTax = Math.round((cgst + sgst + igst) * 100) / 100;
  const beforeRound = Math.round((taxable + totalTax) * 100) / 100;
  const grandTotal = Math.round(beforeRound);
  const roundOff = Math.round((grandTotal - beforeRound) * 100) / 100;

  const build = function(keyFn) {
    const map = {}, order = [];
    lines.forEach(function(l) {
      const k = keyFn(l);
      if (!map[k]) { map[k] = { hsn: l.hsn, gstPct: l.gstPct, taxable: 0, cgst: 0, sgst: 0, igst: 0 }; order.push(k); }
      map[k].taxable += l.taxable; map[k].cgst += l.cgst; map[k].sgst += l.sgst; map[k].igst += l.igst;
    });
    return order.map(function(k) { return map[k]; });
  };
  let groups = build(function(l) { return l.hsn + '|' + l.gstPct; });
  if (groups.length > 8) {
    groups = build(function(l) { return String(l.gstPct); });
    groups.forEach(function(g) { g.hsn = 'Various'; });
  }
  groups.forEach(function(g) { g.tax = Math.round((g.cgst + g.sgst + g.igst) * 100) / 100; });
  groups.sort(function(a, b) { return a.gstPct - b.gstPct || String(a.hsn).localeCompare(String(b.hsn)); });

  return { items: items, lines: lines, intra: intra, taxable: taxable, cgst: cgst, sgst: sgst, igst: igst,
           totalTax: totalTax, grandTotal: grandTotal, roundOff: roundOff, groups: groups };
}

// EMAILED INVOICE PDF — layout matched to the browser print engine
//
// The browser prints from LedgerPrint (index.html). This function produces the
// PDF that gets emailed. Both must lay out the same invoice the same way, so
// both are driven by the same millimetre constants and the same pagination
// rules, reproduced below.
//
// One thing does NOT carry across: the CSS mechanics. The browser engine uses
// a flex column with mm block heights. This output goes through Google Drive's
// HTML-to-PDF converter, which does not handle flexbox or percentage heights
// reliably, so the same geometry is expressed as plain tables with explicit
// pixel heights converted from the shared mm figures. Numbers are shared;
// only the markup differs.
// ═══════════════════════════════════════════════════════════════════════════

// Shared print geometry, millimetres. Keep in step with GEO in index.html.
var INV_GEO = {
  pageH: 279, pageW: 194,
  titleH: 7, sellerH: 22, partiesH: 28, stripH: 6, contStripH: 12,
  tableHeadH: 6, rowH: 4.8, metaRowH: 3.2,
  taxHeadH: 5.2, taxRowH: 4.6, taxFootH: 5,
  totalsH: 24, wordsH: 9, bankH: 22, pageFootH: 5,
  pageBorder: 0.4, cellPadX: 1.2, descFontPt: 7.5, maxRowUnits: 3
};
var MM_TO_PX = 3.779528;               // 96 dpi, what the converter assumes
function invPx(mm) { return Math.round(mm * MM_TO_PX * 100) / 100; }

// Arial / Helvetica advance widths, units per 1000 em. Line counts are derived
// from these so a long description consumes the same number of row slots here
// as it does in the browser, and the two never split pages differently.
var INV_AFM = (function () {
  var w = { 32:278,33:278,34:355,35:556,36:556,37:889,38:667,39:191,40:333,41:333,
    42:389,43:584,44:278,45:333,46:278,47:278,58:278,59:278,60:584,61:584,62:584,
    63:556,64:1015,91:278,92:278,93:278,94:469,95:556,96:333,123:334,124:260,125:334,126:584 };
  '0123456789'.split('').forEach(function (c) { w[c.charCodeAt(0)] = 556; });
  var up = { A:667,B:667,C:722,D:722,E:667,F:611,G:778,H:722,I:278,J:500,K:667,L:556,
    M:833,N:722,O:778,P:667,Q:778,R:722,S:667,T:611,U:722,V:667,W:944,X:667,Y:667,Z:611 };
  var lo = { a:556,b:556,c:500,d:556,e:556,f:278,g:556,h:556,i:222,j:222,k:500,l:222,
    m:833,n:556,o:556,p:556,q:556,r:333,s:500,t:278,u:556,v:500,w:722,x:500,y:500,z:500 };
  Object.keys(up).forEach(function (c) { w[c.charCodeAt(0)] = up[c]; });
  Object.keys(lo).forEach(function (c) { w[c.charCodeAt(0)] = lo[c]; });
  return w;
})();

function invTextWidthMM(text, fontPt) {
  var s = String(text == null ? '' : text), u = 0;
  for (var i = 0; i < s.length; i++) {
    var c = s.charCodeAt(i);
    u += (INV_AFM[c] === undefined ? 556 : INV_AFM[c]);
  }
  return u / 1000 * fontPt * 0.352778;
}

function invWrappedLineCount(text, widthMM, fontPt) {
  var words = String(text == null ? '' : text).trim().split(/\s+/).filter(function (w) { return w; });
  if (!words.length) return 1;
  var space = invTextWidthMM(' ', fontPt), lines = 1, used = 0;
  for (var i = 0; i < words.length; i++) {
    var w = invTextWidthMM(words[i], fontPt);
    if (w > widthMM) {
      if (used > 0) { lines++; used = 0; }
      var rows = Math.ceil(w / widthMM);
      lines += rows - 1;
      used = w - (rows - 1) * widthMM;
      continue;
    }
    var add = used === 0 ? w : space + w;
    if (used + add > widthMM) { lines++; used = w; } else { used += add; }
  }
  return lines;
}

function invDescWidthMM() {
  return (INV_GEO.pageW - 2 * INV_GEO.pageBorder) * 43 / 100 - 2 * INV_GEO.cellPadX;
}
function invRowUnits(line) {
  return Math.min(INV_GEO.maxRowUnits,
    Math.max(1, invWrappedLineCount(line.desc, invDescWidthMM(), INV_GEO.descFontPt)));
}

// ── page arithmetic, identical to the browser engine ───────────────────────
function invHeaderH(first) {
  return first ? INV_GEO.titleH + INV_GEO.sellerH + INV_GEO.partiesH + INV_GEO.stripH
               : INV_GEO.titleH + INV_GEO.contStripH;
}
function invTaxBlockH(groupCount) {
  return INV_GEO.taxHeadH + groupCount * INV_GEO.taxRowH + INV_GEO.taxFootH;
}
// Every page reserves the same footer band, last page or not: the final page
// carries the tax summary, totals, words, bank and signature; earlier pages
// carry the page subtotals, bank details and a pointer to the final page.
// Identical reservation means a page break never changes the line count.
function invFooterH(groupCount) {
  return invTaxBlockH(groupCount) + INV_GEO.totalsH + INV_GEO.wordsH +
         INV_GEO.bankH + INV_GEO.pageFootH;
}
function invTableAreaH(first, groupCount) {
  return Math.round((INV_GEO.pageH - invHeaderH(first) - invFooterH(groupCount)) * 100) / 100;
}
function invPageCapacity(first, groupCount) {
  return Math.max(1, Math.floor((invTableAreaH(first, groupCount) - INV_GEO.tableHeadH) / INV_GEO.rowH));
}

function invPaginate(lines, groupCount) {
  var units = lines.map(invRowUnits), pages = [], i = 0;
  while (i < lines.length) {
    var first = pages.length === 0;
    var cap = invPageCapacity(first, groupCount);
    var rest = 0;
    for (var k = i; k < lines.length; k++) rest += units[k];
    if (rest <= cap) {
      pages.push({ from: i, items: lines.slice(i), first: first, isLast: true, used: rest });
      i = lines.length;
    } else {
      var used = 0, j = i;
      while (j < lines.length && used + units[j] <= cap) { used += units[j]; j++; }
      if (j === i) { used = units[i]; j = i + 1; }
      pages.push({ from: i, items: lines.slice(i, j), first: first, isLast: false, used: used });
      i = j;
    }
  }
  if (!pages.length) pages.push({ from: 0, items: [], first: true, isLast: true, used: 0 });
  pages.forEach(function (p, idx) {
    p.number = idx + 1;
    p.capacity = invPageCapacity(p.first, groupCount);
    p.tableH = invTableAreaH(p.first, groupCount);
    p.bodyH = Math.round((p.tableH - INV_GEO.tableHeadH) * 100) / 100;
    p.filler = Math.round((p.bodyH - p.used * INV_GEO.rowH) * 100) / 100;
  });
  return pages;
}

// ── formatting ─────────────────────────────────────────────────────────────
function invEsc(v) {
  return String(v == null ? '' : v)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function invGroupIN(n) {
  var neg = n < 0;
  n = Math.abs(Math.round(n * 100) / 100);
  var s = n.toFixed(2), parts = s.split('.'), whole = parts[0];
  var last3 = whole.length > 3 ? whole.slice(-3) : whole;
  var rest = whole.length > 3 ? whole.slice(0, -3) : '';
  if (rest) rest = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',';
  return (neg ? '-' : '') + rest + last3 + '.' + parts[1];
}
function invNum(n) { return invGroupIN(parseFloat(n || 0)); }
function invCur(n) { return '&#x20B9;' + invGroupIN(parseFloat(n || 0)); }
function invDMY(v) {
  var ymd = gasDateToYMD(v);
  if (!ymd) return '-';
  var mon = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  var p = ymd.split('-');
  return p[2] + '-' + mon[parseInt(p[1], 10) - 1] + '-' + p[0];
}
function invFY(v) {
  var ymd = gasDateToYMD(v);
  var d = ymd ? new Date(ymd) : new Date();
  if (isNaN(d.getTime())) d = new Date();
  var y = d.getFullYear(), start = d.getMonth() >= 3 ? y : y - 1;
  return start + '-' + String(start + 1).slice(2);
}

function buildInvoiceHtml(data) {
  var calc = computeInvoiceLineCalc(data);
  var lines = calc.lines;
  var groups = calc.groups;
  var intra = calc.intra;
  var pages = invPaginate(lines, groups.length);
  var totalPages = pages.length;
  var buyerState = data.buyerState || '';
  var buyerCode = data.stateCode || '';
  // numberToWordsGas() already appends "Only" itself (matches the client's
  // numberToIndianWords) — appending it again here produced "...Only Only".
  var amountInWords = 'Rupees ' + numberToWordsGas(Math.round(calc.grandTotal));

  // Columns, same set and widths as the browser engine.
  var COLS = [
    { t: 'S/N', w: 5, a: 'center' },
    { t: 'Description of goods and services', w: 43, a: 'left' },
    { t: 'HSN/SAC', w: 8, a: 'center' },
    { t: 'Unit', w: 6, a: 'center' },
    { t: 'Qty', w: 8, a: 'right' },
    { t: 'Rate', w: 11, a: 'right' },
    { t: 'Taxable value', w: 19, a: 'right' }
  ];

  function itemRowHtml(l, sn) {
    var h = invPx(invRowUnits(l) * INV_GEO.rowH);
    var cells = [
      { v: sn, a: 'center' },
      { v: invEsc(l.desc), a: 'left' },
      { v: invEsc(l.hsn || ''), a: 'center' },
      { v: invEsc(l.unit || 'Nos'), a: 'center' },
      { v: invNum(l.qty), a: 'right' },
      { v: invNum(l.rate), a: 'right' },
      { v: invNum(l.taxable), a: 'right' }
    ];
    return '<tr style="height:' + h + 'px;">' + cells.map(function (c, i) {
      return '<td class="ig' + (i === cells.length - 1 ? ' last' : '') +
        '" style="text-align:' + c.a + ';">' + c.v + '</td>';
    }).join('') + '</tr>';
  }

  function fillerRowHtml(mm) {
    if (mm <= 0.05) return '';
    return '<tr style="height:' + invPx(mm) + 'px;">' + COLS.map(function (c, i) {
      return '<td class="ig' + (i === COLS.length - 1 ? ' last' : '') + '">&nbsp;</td>';
    }).join('') + '</tr>';
  }

  function itemsTableHtml(page) {
    var rows = '';
    page.items.forEach(function (l, i) { rows += itemRowHtml(l, page.from + i + 1); });
    return '<table class="grid" cellpadding="0" cellspacing="0" style="height:' + invPx(page.tableH) + 'px;">' +
      '<colgroup>' + COLS.map(function (c) { return '<col style="width:' + c.w + '%;">'; }).join('') + '</colgroup>' +
      '<thead><tr style="height:' + invPx(INV_GEO.tableHeadH) + 'px;">' +
      COLS.map(function (c, i) {
        return '<th class="ig' + (i === COLS.length - 1 ? ' last' : '') +
          '" style="text-align:' + (c.a === 'right' ? 'right' : c.a) + ';">' + invEsc(c.t) + '</th>';
      }).join('') + '</tr></thead><tbody>' + rows + fillerRowHtml(page.filler) + '</tbody></table>';
  }

  // ── header blocks ────────────────────────────────────────────────────────
  function titleHtml() {
    return '<table class="band" cellpadding="0" cellspacing="0" style="height:' + invPx(INV_GEO.titleH) + 'px;">' +
      '<tr><td class="pad ttl">TAX INVOICE</td>' +
      '<td class="pad copy">ORIGINAL FOR RECIPIENT</td></tr></table>';
  }

  function sellerHtml() {
    function id(k, v) {
      return v ? '<tr><td class="lbl">' + k + '</td><td class="bold rt">' + invEsc(v) + '</td></tr>' : '';
    }
    var addr = String(COMPANY.address || '').split(/\r?\n/).map(invEsc).join('<br>');
    return '<table class="band" cellpadding="0" cellspacing="0" style="height:' + invPx(INV_GEO.sellerH) + 'px;">' +
      '<tr>' +
      (COMPANY.logo ? '<td class="pad" width="52" style="vertical-align:middle;"><img src="' + COMPANY.logo +
        '" style="max-width:46px;max-height:46px;border:0;"></td>' : '') +
      '<td class="pad" style="vertical-align:top;">' +
        '<div class="coname">' + invEsc(COMPANY.name) + '</div>' +
        '<div class="coaddr">' + addr + '</div></td>' +
      '<td class="pad" width="235" style="vertical-align:top;">' +
        '<table cellpadding="0" cellspacing="0" class="ids">' +
        id('GSTIN', COMPANY.gstin) + id('PAN', COMPANY.pan) + id('Phone', COMPANY.phone) +
        id('Email', COMPANY.email) + id('Web', COMPANY.website) +
        '</table></td></tr></table>';
  }

  function partyLinesHtml(name, address, gstin, state, code, phone) {
    var out = [];
    String(address || '').split(/\r?\n/).forEach(function (t) { if (t && t.trim()) out.push(t.trim()); });
    if (gstin) out.push('GSTIN: ' + gstin);
    if (state) out.push('State: ' + state + (code ? ' (code ' + code + ')' : ''));
    if (phone) out.push('Phone: ' + phone);

      '<div class="plines">' + out.map(invEsc).join('<br>') + '</div>';

  function partiesHtml() {
    function metaRow(k, v) {
      return '<tr style="height:' + invPx(INV_GEO.metaRowH) + 'px;">' +
        '<td class="mk">' + invEsc(k) + '</td><td class="mv">' + invEsc(v || '-') + '</td></tr>';
    }
    var pos = buyerState ? buyerState + (buyerCode ? ' (' + buyerCode + ')' : '') : '';
    var meta = metaRow('Invoice no.', data.invoiceNo) +
      metaRow('Invoice date', invDMY(data.date)) +
      metaRow('Challan no.', data.challanNo) +
      metaRow('Place of supply', pos) +
      metaRow('Destination', buyerState);
    // Ship To is deliberately not rendered here — invoices show Bill To
    // only, widened to take the freed 30% (34% + 30% = 64%).
    return '<table class="band" cellpadding="0" cellspacing="0" style="height:' + invPx(INV_GEO.partiesH) + 'px;">' +
      '<tr>' +
      '<td class="pad brt" width="64%" style="vertical-align:top;"><div class="lbl">BILL TO</div>' +
        partyLinesHtml(data.buyerName, data.buyerAddress, data.buyerGstin, buyerState, buyerCode, data.buyerPhone) + '</td>' +
      '<td width="235" style="vertical-align:top;padding:0;">' +
        '<table cellpadding="0" cellspacing="0" class="metatbl">' + meta + '</table></td>' +
      '</tr></table>';
  }

  function stripHtml() {
    var bits = [
      intra ? 'Intra-state supply, CGST and SGST applicable' : 'Inter-state supply, IGST applicable',
      'Currency: INR',
      'FY ' + invFY(data.date)
    ];
    return '<table class="band" cellpadding="0" cellspacing="0" style="height:' + invPx(INV_GEO.stripH) + 'px;">' +
      '<tr>' + bits.map(function (b, i) {
        return '<td class="pad strip" style="text-align:' + (i === 0 ? 'left' : (i === bits.length - 1 ? 'right' : 'center')) + ';">' +
          invEsc(b) + '</td>';
      }).join('') + '</tr></table>';
  }

  function contStripHtml(page) {
    return '<table class="band" cellpadding="0" cellspacing="0" style="height:' + invPx(INV_GEO.contStripH) + 'px;">' +
      '<tr><td class="pad" style="vertical-align:middle;"><span class="bold">' + invEsc(COMPANY.name) + '</span><br>' +
        'Invoice ' + invEsc(data.invoiceNo) + ' dated ' + invDMY(data.date) + '</td>' +
      '<td class="pad" style="text-align:right;vertical-align:middle;">' + invEsc(data.buyerName) + '<br>' +
        'Continued from page ' + (page.number - 1) + '</td></tr></table>';
  }

  // ── footer blocks ────────────────────────────────────────────────────────
  function taxSummaryHtml() {
    var head = intra
      ? ['HSN/SAC','Taxable value','CGST %','CGST amount','SGST %','SGST amount','Total tax']
      : ['HSN/SAC','Taxable value','IGST %','IGST amount','Total tax'];
    var widths = intra ? [22,22,8,16,8,16,8] : [26,28,10,22,14];
    var rows = groups.map(function (g) {
      var cells = intra
        ? [invEsc(g.hsn || '-'), invNum(g.taxable), invNum(g.gstPct / 2), invNum(g.cgst),
           invNum(g.gstPct / 2), invNum(g.sgst), invNum(g.tax)]
        : [invEsc(g.hsn || '-'), invNum(g.taxable), invNum(g.gstPct), invNum(g.igst), invNum(g.tax)];
      return '<tr style="height:' + invPx(INV_GEO.taxRowH) + 'px;">' + cells.map(function (c, i) {
        return '<td class="tx' + (i === cells.length - 1 ? ' last' : '') +

  }

  function totalsHtml() {
    function row(k, v, cls) {
      return '<tr class="' + (cls || '') + '"><td class="sk">' + invEsc(k) + '</td>' +
        '<td class="sv">' + v + '</td></tr>';
    }
    // Rate suffix mirrors the client engine's totals box exactly — reads
    // the invoice's own stored rate (not recomputed from line items, same
    // approximation the client makes), so a genuinely mixed-rate invoice
    // still shows *a* rate rather than none at all.
    var taxRows = intra
      ? row('CGST' + (data.cgstRate ? ' @ ' + data.cgstRate + '%' : ''), invCur(calc.cgst)) +

// ── Number to Words (GAS-compatible, no arrow functions) ─────────────────────

function numberToWordsGas(price) {
  const sglDigit  = ['Zero','One','Two','Three','Four','Five','Six','Seven','Eight','Nine'];
  const dblDigit  = ['Ten','Eleven','Twelve','Thirteen','Fourteen','Fifteen','Sixteen','Seventeen','Eighteen','Nineteen'];
  const tensPlace = ['','Ten','Twenty','Thirty','Forty','Fifty','Sixty','Seventy','Eighty','Ninety'];

  if (!price || price === 0) return 'Zero Only';
  if (price.toString().length > 9) return 'Amount too large';

  const numString   = price.toString().padStart(9, '0');
  const crores      = parseInt(numString.substring(0, 2), 10);
  const lakhs       = parseInt(numString.substring(2, 4), 10);
  const thousands   = parseInt(numString.substring(4, 6), 10);
  const hundreds    = parseInt(numString.substring(6, 7), 10);
  const tensAndOnes = parseInt(numString.substring(7, 9), 10);

  var words = '';

  function proc(num, suffix) {
    if (num <= 0) return;
    if (num < 10) {
      words += sglDigit[num] + ' ' + suffix + ' ';
    } else if (num < 20) {
      words += dblDigit[num - 10] + ' ' + suffix + ' ';
    } else {
      words += tensPlace[Math.floor(num / 10)] + ' ';
      if (num % 10 > 0) words += sglDigit[num % 10] + ' ';
      words += suffix + ' ';
    }
  }

  proc(crores,    'Crore');
  proc(lakhs,     'Lakh');
  proc(thousands, 'Thousand');
  if (hundreds > 0) words += sglDigit[hundreds] + ' Hundred ';
  if (tensAndOnes > 0) {
    if (words !== '') words += 'and ';
    if (tensAndOnes < 10)       words += sglDigit[tensAndOnes] + ' ';
    else if (tensAndOnes < 20)  words += dblDigit[tensAndOnes - 10] + ' ';
    else {
      words += tensPlace[Math.floor(tensAndOnes / 10)] + ' ';
      if (tensAndOnes % 10 > 0) words += sglDigit[tensAndOnes % 10] + ' ';
    }
  }
  return words.trim() + ' Only';
}

// AUTO EMAIL — CHALLAN PDF SENDER
// ═══════════════════════════════════════════════════════════════════════════
//
// Fires automatically after every successful challan save.
// Builds a self-contained inline-CSS HTML challan, converts it to PDF
// via DriveApp temp file method, and sends to fixed SELLER_EMAIL via MailApp.
//
// Called by handleSaveChallan() after sheet.appendRow() succeeds.
// Returns: { sent: true/false, message: string }
// ─────────────────────────────────────────────────────────────────────────

function sendChallanEmail(data) {
  const SELLER_EMAIL = 'itransform007@gmail.com';

  // ── Build challan HTML ───────────────────────────────────────────────────
  const htmlContent = buildChallanHtml(data);

  const safeChallanNo = (data.challanNo || 'Challan').replace(/\//g, '-');
  const safeBuyerName = String(data.buyerName || 'Consignee').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName   = safeBuyerName + ' - ' + safeChallanNo + '.pdf';

  // ── Convert HTML → PDF via DriveApp temp file ───────────────────────────
  var tempFile = null;
  var pdfBlob;
  try {
    tempFile = DriveApp.createFile(
      Utilities.newBlob(htmlContent, 'text/html', 'temp_challan.html')
    );
    pdfBlob = tempFile.getAs('application/pdf');
    pdfBlob.setName(pdfFileName);
  } finally {
    if (tempFile) {
      try { tempFile.setTrashed(true); } catch(e) { Logger.log('Temp file cleanup: ' + e); }
    }
  }

  // ── Format email subject & body ─────────────────────────────────────────
  const subject = 'New Challan Saved: ' + (data.challanNo || '') + ' | ' + (data.buyerName || 'Unknown Consignee');

  const rawDate = gasDateToYMD(data.date);
  const fmtDate = rawDate ? rawDate.split('-').reverse().join('-') : '-';

  const emailBody =
    'A new delivery challan has been saved to Google Sheets.\n\n' +
    'Challan Details:\n' +
    '  Challan No.   : ' + (data.challanNo || '-') + '\n' +
    '  Date          : ' + fmtDate + '\n' +
    '  Consignee     : ' + (data.buyerName || '-') + '\n' +
    '  Buyer GSTIN   : ' + (data.buyerGstin || 'Unregistered') + '\n' +
    (data.refInvoiceNo ? '  Ref Invoice   : ' + data.refInvoiceNo + '\n' : '') +
    '\nPlease find the delivery challan PDF attached.\n\n' +
    '-- Enterprises Challan System';

  // ── Send email to seller ────────────────────────────────────────────────
  MailApp.sendEmail(SELLER_EMAIL, subject, emailBody, {
    name:        'Enterprises — Challan System',
    attachments: [pdfBlob]
  });

  Logger.log('Challan email sent to ' + SELLER_EMAIL + ' for ' + (data.challanNo || ''));
  return { sent: true, message: 'Email sent to ' + SELLER_EMAIL };
}

// BUYER-FACING INVOICE SENDING — WhatsApp link + direct email to the buyer

// Distinct from sendInvoiceEmail() above, which always notifies the fixed

// Finds an invoice row by number and returns enough to both read and, if
// needed, write back to it (e.g. caching the generated PDF URL).
function findInvoiceRow(invoiceNo) {
  const sheet = getSheet(SHEET_INVOICES);
  const allData = sheet.getDataRange().getValues();
  const headers = allData[0].map(function(h) { return String(h).trim(); });
  for (var i = 1; i < allData.length; i++) {
    if ((allData[i][0] || '').toString().trim().toLowerCase() === invoiceNo.toLowerCase()) {
      return { sheet: sheet, headers: headers, rowIdx: i, row: allData[i] };
    }
  }
  return null;
}

// Reconstructs the same payload shape the LedgerPrintGas port expects,
// straight from the sheet row — mirrors handleEmailInvoiceReport's mapping
// below, plus buyerState/stateCode/shipTo/taxMode so a buyer-facing PDF gets
// the correct intra vs inter-state GST split. Includes subTotal/CGST/SGST/
// IGST AMOUNTS (not just the rates) — the ported engine trusts these stored
// figures rather than recomputing them, so without them here it would print
// ₹0.00 for taxable value/CGST/SGST despite a correct grand total.
function buildInvoicePayloadFromRow(headers, row, invoiceNo) {
  var inv = {};
  headers.forEach(function(h, j) { inv[h] = row[j]; });
  var items; try { items = JSON.parse(inv['Items (JSON)'] || '[]'); } catch (e) { items = []; }
  var shipToEnvelope = {};
  try { shipToEnvelope = inv['Ship To (JSON)'] ? JSON.parse(inv['Ship To (JSON)']) : {}; } catch (e) { shipToEnvelope = {}; }
  return {
    invoiceNo: inv['Invoice No'] || invoiceNo, date: inv['Date'] || '', challanNo: inv['Challan No'] || '',
    buyerName: inv['Buyer Name'] || '', buyerPhone: inv['Buyer Phone'] || '', buyerEmail: inv['Buyer Email'] || '',
    buyerAddress: inv['Buyer Address'] || '', buyerGstin: inv['Buyer GSTIN'] || '',
    buyerState: inv['Buyer State'] || '', stateCode: inv['State Code'] || '',
    grandTotal: inv['Grand Total'] || 0, items: items,
    subTotal: inv['Sub Total'] || 0,
    cgstRate: inv['CGST Rate'] || 0, sgstRate: inv['SGST Rate'] || 0, igstRate: inv['IGST Rate'] || 0,
    cgstAmount: inv['CGST Amount'] || 0, sgstAmount: inv['SGST Amount'] || 0, igstAmount: inv['IGST Amount'] || 0,
    shipTo: shipToEnvelope.sameShip === false ? (shipToEnvelope.shipTo || null) : null,
    taxMode: shipToEnvelope.taxMode || 'auto'
  };
}

// Builds the invoice PDF and saves it to Drive, shared "anyone with the
// link" — unlike sendInvoiceEmail's temp-file-then-trash pattern, this copy
// must persist so the WhatsApp message's link keeps working after the
// request ends.
function saveInvoicePdfToDrive(data) {
  const safeInvoiceNo = (data.invoiceNo || 'Invoice').replace(/\//g, '-');
  const safeBuyerName = String(data.buyerName || 'Buyer').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName = safeBuyerName + ' - ' + safeInvoiceNo + '.pdf';

  // Built by the LedgerPrintGas port of the browser's print engine — see
  // that module's header comment (a few hundred lines above) for why.
  const pdfBlob = renderLedgerPrintPdf(data, COMPANY, 'invoice', pdfFileName);

  const folder = getOrCreateAppFolder('GME InvoiceApp - Shared Invoice PDFs');
  const pdfFile = folder.createFile(pdfBlob);
  pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return pdfFile.getUrl();
}

// Emails the invoice PDF straight to the buyer (not the seller) — a
// buyer-facing subject/body, distinct from sendInvoiceEmail()'s internal
// "a new invoice was saved" notification above.
function sendInvoiceCopyToBuyer(data) {
  const toEmail = (data.buyerEmail || '').toString().trim();
  if (!toEmail) return { sent: false, message: 'No buyer email on file for this invoice.' };

  const safeInvoiceNo = (data.invoiceNo || 'Invoice').replace(/\//g, '-');
  const safeBuyerName = String(data.buyerName || 'Buyer').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName = safeBuyerName + ' - ' + safeInvoiceNo + '.pdf';

  // Built by the LedgerPrintGas port of the browser's print engine — see
  // that module's header comment (a few hundred lines above) for why.
  const pdfBlob = renderLedgerPrintPdf(data, COMPANY, 'invoice', pdfFileName);

  const subject = 'Invoice ' + (data.invoiceNo || '') + ' from ' + (COMPANY.name || '');
  const grandTotalFormatted = 'Rs.' + parseFloat(data.grandTotal || 0).toFixed(2);
  const emailBody =
    'Dear ' + (data.buyerName || 'Customer') + ',\n\n' +
    'Please find your invoice attached.\n\n' +
    '  Invoice No.  : ' + (data.invoiceNo || '-') + '\n' +
    '  Grand Total  : ' + grandTotalFormatted + '\n\n' +
    'Thank you for your business.\n\n' +
    'Regards,\n' + (COMPANY.name || '');

  MailApp.sendEmail(toEmail, subject, emailBody, {
    name: COMPANY.name || 'Invoice',
    attachments: [pdfBlob]
  });

  return { sent: true, message: 'Invoice emailed to ' + toEmail };
}

function handleSendInvoiceToBuyer(body) {
  const invoiceNo = (body.invoiceNo || '').toString().trim();
  if (!invoiceNo) return jsonOut({ success: false, message: 'Invoice No. is required' });
  const found = findInvoiceRow(invoiceNo);
  if (!found) return jsonOut({ success: false, message: 'Invoice not found: ' + invoiceNo });
  const payload = buildInvoicePayloadFromRow(found.headers, found.row, invoiceNo);
  try {
    const result = sendInvoiceCopyToBuyer(payload);
    return jsonOut({ success: result.sent, message: result.message });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// Returns { phone, buyerName, pdfUrl } for the frontend to build a wa.me
// click-to-chat link from. Regenerates the PDF fresh on every call (see the
// comment inside) and overwrites the cached "Invoice PDF URL" column with
// the new file's link.
function handleGetInvoiceWhatsappLink(body) {
  const invoiceNo = (body.invoiceNo || '').toString().trim();
  if (!invoiceNo) return jsonOut({ success: false, message: 'Invoice No. is required' });
  const found = findInvoiceRow(invoiceNo);
  if (!found) return jsonOut({ success: false, message: 'Invoice not found: ' + invoiceNo });

  const phoneCol = found.headers.indexOf('Buyer Phone');
  const phone = (phoneCol > -1 ? found.row[phoneCol] : '').toString().trim();
  if (!phone) return jsonOut({ success: false, message: 'No buyer phone on file for this invoice.' });

  // Always regenerate — a cached URL here would keep serving a PDF built
  // under old branding (logo/signature/stamp) or an old layout fix forever,
  // since neither ever changes this invoice's own row to invalidate it.
  // Email already regenerates fresh on every send; WhatsApp now matches
  // that instead of trying to reuse a stale file.
  const urlCol = found.headers.indexOf('Invoice PDF URL');
  var pdfUrl;
  const payload = buildInvoicePayloadFromRow(found.headers, found.row, invoiceNo);
  try {
    pdfUrl = saveInvoicePdfToDrive(payload);
    if (urlCol > -1) found.sheet.getRange(found.rowIdx + 1, urlCol + 1).setValue(pdfUrl);
  } catch (err) {
    return jsonOut({ success: false, message: 'Could not generate invoice PDF: ' + err.toString() });
  }

  const buyerNameCol = found.headers.indexOf('Buyer Name');
  return jsonOut({
    success: true,
    phone: phone,
    buyerName: buyerNameCol > -1 ? (found.row[buyerNameCol] || '') : '',
    pdfUrl: pdfUrl
  });
}

// QUOTATION HANDLERS
// ═══════════════════════════════════════════════════════════════════════════
//
// Sheet: "Quotations"
// Columns: Quotation No | Date | Valid Until | Ref No | Buyer Name | Buyer Phone |
//          Buyer Email | Buyer Address | Buyer GSTIN | Buyer State |
//          Sub Total | Tax Total | Grand Total | Notes | Items (JSON) | Saved By | Timestamp
//
// Auto-increment format: QT/YY-YY/NNN  e.g. QT/26-27/001
// ─────────────────────────────────────────────────────────────────────────

function handleGetNextQuotationNo() {
  try {
    const sheet = getSheet(SHEET_QUOTATIONS);
    const data  = sheet.getDataRange().getValues();

    const existingNos = data.slice(1)
      .map(r => (r[0] || '').toString().trim())
      .filter(v => v !== '');

    const now     = new Date();
    const yr      = now.getFullYear();
    const mo      = now.getMonth() + 1;
    const fyStart = mo >= 4 ? yr : yr - 1;
    const fyEnd   = fyStart + 1;
    const fySuffix = String(fyStart).slice(-2) + '-' + String(fyEnd).slice(-2);
    const prefix   = 'QT/' + fySuffix + '/';

    let maxSeq = 0;
    existingNos.forEach(function(no) {
      if (no.toUpperCase().startsWith(prefix.toUpperCase())) {
        const parts = no.split('/');
        const seq   = parseInt(parts[parts.length - 1], 10);
        if (!isNaN(seq) && seq > maxSeq) maxSeq = seq;
      }
    });

    const nextSeq  = maxSeq + 1;
    const quotNo   = prefix + String(nextSeq).padStart(3, '0');
    return jsonOut({ success: true, quotNo: quotNo });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

}

function handleSaveQuotation(body, username) {
  const sheet   = getSheet(SHEET_QUOTATIONS);
  const allData = sheet.getDataRange().getValues();
  const quotNo  = (body.quotNo || '').toString().trim();

  if (!quotNo) return jsonOut({ success: false, message: 'Quotation No. is required' });

  const exists = allData.slice(1).some(function(r) {
    return (r[0] || '').toString().trim().toLowerCase() === quotNo.toLowerCase();
  });
  if (exists) return jsonOut({ success: false, message: 'Duplicate quotation number — ' + quotNo + ' already exists' });

  const now = new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' });

  sheet.appendRow([
    quotNo,
    body.date        || '',
    body.validUntil  || '',
    body.refNo       || '',
    body.buyerName   || '',
    body.buyerPhone  || '',
    body.buyerEmail  || '',
    body.buyerAddress || '',
    body.buyerGstin  || '',
    body.buyerState  || '',
    parseFloat(body.subTotal   || 0),
    parseFloat(body.taxTotal   || 0),
    parseFloat(body.grandTotal || 0),
    body.notes       || '',
    body.itemsJson   || '[]',
    username,
    now
  ]);

  // ── AUTO EMAIL ────────────────────────────────────────────────────────────
  var emailSent    = false;
  var emailMessage = '';
  try {
    var emailPayload = {
      quotNo:      quotNo,
      date:        body.date        || '',
      validUntil:  body.validUntil  || '',
      refNo:       body.refNo       || '',
      buyerName:   body.buyerName   || '',
      buyerPhone:  body.buyerPhone  || '',
      buyerEmail:  body.buyerEmail  || '',
      buyerAddress: body.buyerAddress || '',
      buyerGstin:  body.buyerGstin  || '',
      buyerState:  body.buyerState  || '',
      subTotal:    parseFloat(body.subTotal   || 0),
      taxTotal:    parseFloat(body.taxTotal   || 0),
      grandTotal:  parseFloat(body.grandTotal || 0),
      notes:       body.notes       || '',
      items:       JSON.parse(body.itemsJson || '[]')
    };
    var emailResult = sendQuotationEmail(emailPayload);
    emailSent    = emailResult.sent;
    emailMessage = emailResult.message;
  } catch (emailErr) {
    emailMessage = emailErr.toString();
    Logger.log('Quotation email error: ' + emailMessage);
  }

  return jsonOut({ success: true, emailSent: emailSent, emailMessage: emailMessage });
}

// AUTO EMAIL — QUOTATION PDF SENDER
// ═══════════════════════════════════════════════════════════════════════════

function sendQuotationEmail(data) {
  const SELLER_EMAIL = 'itransform007@gmail.com';

  const htmlContent = buildQuotationHtmlGas(data);

  const safeQuotNo    = (data.quotNo    || 'Quotation').replace(/\//g, '-');
  const safeBuyerName = String(data.buyerName || 'Customer').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName   = safeBuyerName + ' - ' + safeQuotNo + '.pdf';

  var tempFile = null;
  var pdfBlob;
  try {
    tempFile = DriveApp.createFile(
      Utilities.newBlob(htmlContent, 'text/html', 'temp_quotation.html')
    );
    pdfBlob = tempFile.getAs('application/pdf');
    pdfBlob.setName(pdfFileName);
  } finally {
    if (tempFile) {
      try { tempFile.setTrashed(true); } catch(e) { Logger.log('Temp file cleanup: ' + e); }
    }
  }

  const rawDate = gasDateToYMD(data.date);
  const fmtDate = rawDate ? rawDate.split('-').reverse().join('-') : '-';
  const subject = 'New Quotation Saved: ' + (data.quotNo || '') + ' | ' + (data.buyerName || 'Unknown Customer');

  const emailBody =
    'A new quotation has been saved to Google Sheets.\n\n' +
    'Quotation Details:\n' +
    '  Quotation No. : ' + (data.quotNo    || '-') + '\n' +
    '  Date          : ' + fmtDate + '\n' +
    '  Customer      : ' + (data.buyerName || '-') + '\n' +
    '  Grand Total   : Rs.' + parseFloat(data.grandTotal || 0).toFixed(2) + '\n' +
    (data.validUntil ? '  Valid Until   : ' + data.validUntil + '\n' : '') +
    '\nPlease find the quotation PDF attached.\n\n' +
    '-- Enterprises Quotation System';

  MailApp.sendEmail(SELLER_EMAIL, subject, emailBody, {
    name:        'Enterprises — Quotation System',
    attachments: [pdfBlob]
  });

  Logger.log('Quotation email sent to ' + SELLER_EMAIL + ' for ' + (data.quotNo || ''));
  return { sent: true, message: 'Email sent to ' + SELLER_EMAIL };
}

// ─────────────────────────────────────────────────────────────────────────
// BUILD QUOTATION HTML (GAS-side, inline-CSS, A4 portrait)
// ─────────────────────────────────────────────────────────────────────────

function buildQuotationHtmlGas(data) {
  var items     = data.items || [];
  var subTotal  = parseFloat(data.subTotal  || 0);
  var taxTotal  = parseFloat(data.taxTotal  || 0);
  var grandTotal = parseFloat(data.grandTotal || 0);

  var rawDate  = gasDateToYMD(data.date);
  var fmtDate  = rawDate ? rawDate.split('-').reverse().join('-') : '-';
  var rawValid = gasDateToYMD(data.validUntil);
  var fmtValid = rawValid ? rawValid.split('-').reverse().join('-') : '-';

  // Item rows
  var itemRowsHtml = '';
  items.forEach(function(item, idx) {
    var amt  = parseFloat(item.amount || 0).toFixed(2);
    var rate = parseFloat(item.rate   || 0).toFixed(2);
    itemRowsHtml +=
      '<tr style="height:22px;">' +
        '<td style="border-right:1px solid #000;text-align:center;">' + (idx + 1) + '</td>' +
        '<td style="border-right:1px solid #000;padding:2px 5px;"><strong>' + (item.desc || '') + '</strong></td>' +
        '<td style="border-right:1px solid #000;text-align:center;font-family:monospace;">' + (item.hsn || '') + '</td>' +
        '<td style="border-right:1px solid #000;text-align:right;padding-right:4px;font-family:monospace;"><strong>' + (item.qty || '') + '</strong></td>' +
        '<td style="border-right:1px solid #000;text-align:right;padding-right:4px;font-family:monospace;">' + rate + '</td>' +
        '<td style="border-right:1px solid #000;text-align:center;">' + (item.unit || 'Nos') + '</td>' +
        '<td style="border-right:1px solid #000;text-align:center;">' + (item.gstPct || 0) + '%</td>' +
        '<td style="text-align:right;padding-right:4px;font-family:monospace;"><strong>' + amt + '</strong></td>' +
      '</tr>';
  });

  // Tax summary rows grouped by GST %
  var taxMap = {};
  items.forEach(function(i) {
    var p = parseFloat(i.gstPct) || 0;
    if (p > 0) { var t = (i.amount || 0) * p / 100; taxMap[p] = (taxMap[p] || 0) + t; }
  });
  var taxRowsHtml = '';
  Object.keys(taxMap).forEach(function(pct) {
    taxRowsHtml +=
      '<tr style="height:20px;font-size:10px;color:#444;">' +
        '<td style="border-right:1px solid #000;"></td>' +
        '<td colspan="5" style="border-right:1px solid #000;text-align:right;padding-right:4px;font-style:italic;">GST @ ' + pct + '%</td>' +
        '<td style="border-right:1px solid #000;"></td>' +
        '<td style="text-align:right;padding-right:4px;font-family:monospace;">' + parseFloat(taxMap[pct]).toFixed(2) + '</td>' +
      '</tr>';
  });

  // Blank rows
  var blankCount = Math.max(0, 23 - items.length - Object.keys(taxMap).length);

function handleGetNextCreditDebitNoteNo(noteType) {
  try {
    const type = (noteType || 'CREDIT').toString().toUpperCase() === 'DEBIT' ? 'DEBIT' : 'CREDIT';
    const sheet = getSheet(SHEET_CREDIT_DEBIT_NOTES);
    const data  = sheet.getDataRange().getValues();

    const existingNos = data.slice(1)
      .filter(r => (r[0] || '').toString().trim().toUpperCase() === type)
      .map(r => (r[1] || '').toString().trim())
      .filter(v => v !== '');

    const now     = new Date();
    const yr      = now.getFullYear();
    const mo      = now.getMonth() + 1;
    const fyStart = mo >= 4 ? yr : yr - 1;
    const fyEnd   = fyStart + 1;
    const fySuffix = String(fyStart).slice(-2) + '-' + String(fyEnd).slice(-2);
    const prefix   = (type === 'DEBIT' ? 'DN/' : 'CN/') + fySuffix + '/';

    let maxSeq = 0;
    existingNos.forEach(function(no) {
      if (no.toUpperCase().startsWith(prefix.toUpperCase())) {
        const parts = no.split('/');
        const seq   = parseInt(parts[parts.length - 1], 10);
        if (!isNaN(seq) && seq > maxSeq) maxSeq = seq;
      }
    });

    const nextSeq = maxSeq + 1;
    const noteNo  = prefix + String(nextSeq).padStart(3, '0');
    return jsonOut({ success: true, noteNo: noteNo });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

function handleGetCreditDebitNotesData() {
  const sheet   = getSheet(SHEET_CREDIT_DEBIT_NOTES);
  const data    = sheet.getDataRange().getValues();
  const headers = data[0];
  const notes = data.slice(1).map(function(row) {
    var n = {};
    headers.forEach(function(h, j) { n[h] = row[j]; });
    n['Note Type'] = n['Note Type'] || row[0] || '';
    n['Note No']   = n['Note No']   || row[1] || '';
    return n;
  });
  return jsonOut({ notes: notes });
}

function handleGetCreditDebitNoteByNo(noteNo) {
  const sheet   = getSheet(SHEET_CREDIT_DEBIT_NOTES);
  const data    = sheet.getDataRange().getValues();
  const headers = data[0];
  for (var i = 1; i < data.length; i++) {
    if ((data[i][1] || '').toString().trim().toLowerCase() === noteNo.toLowerCase()) {
      var n = {};
      headers.forEach(function(h, j) { n[h] = data[i][j]; });
      n['Note Type'] = n['Note Type'] || data[i][0] || '';
      n['Note No']   = n['Note No']   || data[i][1] || '';
      return jsonOut({ note: n });
    }
  }
  return jsonOut({ note: null });
}

/**
 * Returns all AMC contracts from the AMC_Contracts sheet.
 */
function handleGetAMCData() {
  try {
    const sheet = getSheet(SHEET_AMC);
    const data  = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonOut({ success: true, contracts: [] });
    const headers = data[0].map(h => String(h).trim());
    const contracts = data.slice(1).map(function(row) {
      var obj = {};
      headers.forEach(function(h, i) {
        var val = row[i];
        if (val instanceof Date) val = gasDateToYMD(val);
        obj[h] = val !== undefined && val !== null ? String(val) : '';
      });

    }).filter(function(c) { return c['Contract No'] && c['Customer Name']; });
    return jsonOut({ success: true, contracts: contracts });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// ── AMC: Drive photo storage + Google Calendar reminder helpers ────────────

// Finds (or lazily creates) a Drive folder by name — shared by the AMC site
// photo and the WhatsApp-shared invoice PDF, both of which need somewhere
// persistent to live (unlike the temp-file-then-trash pattern used for the
// seller-notification PDFs above, these must survive after the request ends).
function getOrCreateAppFolder(name) {
  const folders = DriveApp.getFoldersByName(name);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(name);
}

// Decodes a base64 photo (data-URL prefix optional) captured/compressed in
// the browser and stores it in Drive, shared "anyone with the link", so the
// AMC sheet only needs to hold a URL instead of a huge inline cell value.
function saveAmcImageToDrive(base64, contractNo, mimeType) {
  const raw = base64.indexOf(',') > -1 ? base64.split(',')[1] : base64;
  const ext = ((mimeType || 'image/jpeg').split('/')[1] || 'jpg').replace('jpeg', 'jpg');
  const safeNo = (contractNo || 'AMC').replace(/\//g, '-');
  const fileName = 'AMC - ' + safeNo + ' - ' + new Date().getTime() + '.' + ext;
  const blob = Utilities.newBlob(Utilities.base64Decode(raw), mimeType || 'image/jpeg', fileName);
  const folder = getOrCreateAppFolder('GME InvoiceApp - AMC Photos');
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  // Neither file.getUrl() (a viewer *page*, not image bytes) nor the old
  // uc?export=view trick (increasingly intercepted by Google with an
  // interstitial instead of serving the file) render reliably in an <img>
  // tag. Drive's dedicated thumbnail-proxy endpoint is built for exactly
  // this and consistently returns the actual image.
  return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1000';
}

// ── ONE-TIME MAINTENANCE: repair old AMC "Image URL" links ─────────────────
// Rows saved before the thumbnail-proxy fix above hold either Drive's
// viewer-page URL or the old uc?export=view form, neither of which render
// in an <img> tag. This rewrites every AMC "Image URL" cell in place to the
// working thumbnail format, reusing the SAME already-uploaded file — no
// re-upload needed. Not wired to any web action; run it once manually from
// the Apps Script editor (pick this function in the toolbar dropdown, then
// Run) after deploying the fix above. Safe to re-run.
function fixAmcImageUrls() {
  const sheet = getSheet(SHEET_AMC);
  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(function(h) { return String(h).trim(); });
  const col = headers.indexOf('Image URL');
  if (col === -1) { Logger.log('No Image URL column found.'); return; }

  var fixed = 0;
  for (var i = 1; i < data.length; i++) {
    const url = String(data[i][col] || '');
    if (!url) continue;
    const m = url.match(/[-\w]{25,}/); // Drive file IDs are long alphanumeric/-/_ strings
    if (!m) continue;
    const newUrl = 'https://drive.google.com/thumbnail?id=' + m[0] + '&sz=w1000';
    if (newUrl !== url) {
      sheet.getRange(i + 1, col + 1).setValue(newUrl);
      fixed++;
    }
  }
  Logger.log('Fixed ' + fixed + ' AMC image URL(s).');
}

// Creates or updates a reminder dated on the AMC contract's own expiry date
// (Settings → "Google Calendar ID"), with a popup notification attached 30
// days ahead of it. Returns the event ID to store back on the row so the
// NEXT save can find and update the same event instead of creating a
// duplicate. Returns '' if no calendar is configured, the expiry
// date is missing/invalid, or the calendar call fails for any reason — the
// caller treats all of these as "skip", never as a save error.
// Returns { eventId, status } instead of just an ID/blank string, so the
// caller (and ultimately the user, via a toast) can always tell WHY nothing
// happened — "no calendar configured" and "actually failed" used to look
// identical (both silent), which is exactly what made this hard to debug.
// forceToday=true bypasses the expiry-date math entirely (and the
// requirement that an expiry date even be filled in) — used by the manual
// "Add to Calendar" button, which should always land the event on today's
// date on demand, irrespective of what the expiry date says.
function upsertAmcCalendarReminder(d, existingEventId, forceToday) {
  const calId = COMPANY.calendarId || '';
  if (!calId) return { eventId: '', status: 'skipped-no-calendar' };
  if (!d.expiryDate && !forceToday) return { eventId: '', status: 'skipped-no-expiry' };

  const today = new Date(); today.setHours(0, 0, 0, 0);
  var reminderDate;
  if (forceToday) {
    reminderDate = today;
  } else {
    const expiry = new Date(d.expiryDate);
    if (isNaN(expiry.getTime())) return { eventId: '', status: 'skipped-invalid-date' };
    reminderDate = expiry;
    // A contract whose expiry date has already passed would otherwise get a
    // reminder dated in the past — technically created, but sitting
    // somewhere in the calendar nobody would ever scroll back to see. Pin
    // it to today instead so there's always an actual, visible heads-up.
    if (reminderDate < today) reminderDate = today;
  }

  const title = 'AMC Renewal Due: ' + (d.contractNo || '') + ' — ' + (d.customerName || '');
  const description = 'AMC Contract ' + (d.contractNo || '') + ' for ' + (d.customerName || '') +
    ' expires on ' + (d.expiryDate || 'N/A') + '.\nService: ' + (d.service || '') +
    '\nValue: Rs.' + (d.value || 0) + '\nPhone: ' + (d.phone || '') + '\nEmail: ' + (d.email || '');

  // getCalendarById() throws (not returns null) for a bad ID or a calendar
  // this script's account can't see — that propagates up to the caller's
  // try/catch as a real error, which is exactly what we want surfaced.
  const cal = CalendarApp.getCalendarById(calId);
  if (!cal) throw new Error('Calendar "' + calId + '" was not found or is not accessible to this app.');

  var event = null;
  if (existingEventId) {
    try { event = cal.getEventById(existingEventId); } catch (e) { event = null; }
  }
  // A popup notification 30 days ahead of the event itself — separate from
  // where the event is dated (on the expiry date). removeAllReminders()
  // first so re-saving/editing a contract updates this one reminder in
  // place instead of stacking a duplicate on top each time.
  const THIRTY_DAYS_MIN = 30 * 24 * 60;
  if (event) {
    event.setTitle(title);
    event.setDescription(description);
    event.setAllDayDate(reminderDate);
    event.removeAllReminders();
    event.addPopupReminder(THIRTY_DAYS_MIN);
    return { eventId: event.getId(), status: 'updated' };
  }
  const created = cal.createAllDayEvent(title, reminderDate, { description: description });
  created.removeAllReminders();
  created.addPopupReminder(THIRTY_DAYS_MIN);
  return { eventId: created.getId(), status: 'created' };
}

// Manual "Add to Calendar" button in the AMC form — syncs straight from
// whatever is currently typed in, on demand, without going through a full
// contract save. Works for an already-saved contract (rowIndex >= 0, event
// ID persisted back onto its row) or one still being filled in (rowIndex
// -1: the event is created but there's nowhere to store its ID yet, so the
// frontend caches it and forwards it as existingEventId on the real save
// right after, so that save updates this same event instead of duplicating it).
function handleSyncAmcCalendarNow(body) {
  const d = body.data || {};
  if (!d.contractNo || !d.customerName) {
    return jsonOut({ success: false, message: 'Contract No. and Customer Name are required first.' });
  }
  try {
    const result = upsertAmcCalendarReminder(d, body.existingEventId || '', true);
    const rowIndex = parseInt(body.rowIndex, 10);
    if (rowIndex >= 0) {
      const sheet = getSheet(SHEET_AMC);
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function(h) { return String(h).trim(); });
      const evCol = headers.indexOf('Calendar Event ID');
      if (evCol > -1) sheet.getRange(rowIndex + 2, evCol + 1).setValue(result.eventId || '');
    }
    return jsonOut({ success: true, calendarStatus: result.status, eventId: result.eventId || '' });
  } catch (err) {
    return jsonOut({ success: false, calendarStatus: 'error', message: err.toString() });
  }
}

/**
 * Save (add or update) an AMC contract.
 * body.data = { contractNo, customerName, phone, email, service, startDate,
 *   expiryDate, value, freq, notes, location, imageUrl,
 *   imageBase64?, imageMimeType? }
 * body.rowIndex = row index (0-based, excluding header) if updating, or -1 if adding new.
 */
function handleSaveAMCContract(body, username) {
  try {
    const d     = body.data || {};
    const idx   = parseInt(body.rowIndex, 10); // -1 = new
    const sheet = getSheet(SHEET_AMC);
    const ts    = new Date().toISOString();
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function(h) { return String(h).trim(); });
    const evCol = headers.indexOf('Calendar Event ID');

    // A new photo (base64 from the browser) replaces the old one; otherwise
    // keep whatever URL the row already had.
    var imageUrl = d.imageUrl || '';
    if (d.imageBase64) {
      imageUrl = saveAmcImageToDrive(d.imageBase64, d.contractNo || 'AMC', d.imageMimeType || 'image/jpeg');
    }

    const fields = {
      'Contract No': d.contractNo || '', 'Customer Name': d.customerName || '',
      'Phone': d.phone || '', 'Email': d.email || '', 'Service Description': d.service || '',
      'Start Date': d.startDate || '', 'Expiry Date': d.expiryDate || '', 'AMC Value': d.value || '',
      'Payment Frequency': d.freq || 'Annual', 'Notes': d.notes || '',
      'Image URL': imageUrl, 'Location': d.location || '',
      'Saved By': username, 'Timestamp': ts
    };
    // 'Calendar Event ID' is deliberately left out of `fields` — it's only
    // ever written by the calendar upsert below, never clobbered by this map.

    // For a brand-new contract there's no row yet to read a Calendar Event
    // ID from — but the "Add to Calendar" button may have already created
    // one before this save, in which case the frontend forwards it here so
    // it gets updated in place instead of duplicated.
    var sheetRow, existingEventId = body.existingEventId || '';
    if (idx >= 0) {
      // Update existing row (idx is 0-based data row, add 2 for header + 1-based)
      sheetRow = idx + 2;
      if (evCol > -1) existingEventId = sheet.getRange(sheetRow, evCol + 1).getValue() || existingEventId;
      headers.forEach(function(h, i) {
        if (fields.hasOwnProperty(h)) sheet.getRange(sheetRow, i + 1).setValue(fields[h]);
      });
    } else {
      sheet.appendRow(headers.map(function(h) { return fields.hasOwnProperty(h) ? fields[h] : ''; }));
      sheetRow = sheet.getLastRow();
    }

    // Best-effort calendar reminder — a missing/misconfigured tenant
    // calendar must never block saving the contract itself. Every outcome
    // (created/updated/skipped-for-a-named-reason/failed) is reported back
    // so the user always sees SOMETHING instead of ambiguous silence.
    var calendarWarning = '';
    var calendarStatus = 'skipped-no-column';
    if (evCol > -1) {
      try {
        const result = upsertAmcCalendarReminder(d, existingEventId);
        calendarStatus = result.status;
        sheet.getRange(sheetRow, evCol + 1).setValue(result.eventId || '');
      } catch (calErr) {
        calendarStatus = 'error';
        calendarWarning = calErr.toString();
        Logger.log('AMC calendar reminder failed: ' + calendarWarning);
      }
    }

    return jsonOut({
      success: true, message: 'AMC contract saved.',
      calendarStatus: calendarStatus, calendarWarning: calendarWarning
    });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

/**
 * Delete an AMC contract by contract number, and its calendar reminder if any.
 */
function handleDeleteAMCContract(body, username) {
  try {
    const contractNo = (body.contractNo || '').trim();
    if (!contractNo) return jsonOut({ success: false, message: 'Contract No is required.' });
    const sheet = getSheet(SHEET_AMC);
    const data  = sheet.getDataRange().getValues();
    const headers = data[0].map(function(h) { return String(h).trim(); });
    const evCol = headers.indexOf('Calendar Event ID');
    for (var i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0]).trim() === contractNo) {
        const eventId = evCol > -1 ? data[i][evCol] : '';
        if (eventId && COMPANY.calendarId) {
          try {
            const cal = CalendarApp.getCalendarById(COMPANY.calendarId);
            const ev = cal && cal.getEventById(eventId);
            if (ev) ev.deleteEvent();
          } catch (calErr) {
            Logger.log('AMC calendar cleanup failed: ' + calErr);
          }
        }
        sheet.deleteRow(i + 1);
        return jsonOut({ success: true, message: 'Contract deleted.' });
      }
    }
    return jsonOut({ success: false, message: 'Contract not found.' });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

const SHEET_CERTIFICATES = 'Certificates';

function handleGetNextCertNo() {
  try {
    const sheet = getSheet(SHEET_CERTIFICATES);
    const data  = sheet.getDataRange().getValues();

    const existingNos = data.slice(1)
      .map(r => (r[0] || '').toString().trim())
      .filter(v => v !== '');

    const now     = new Date();
    const yr      = now.getFullYear();
    const mo      = now.getMonth() + 1;
    const fyStart = mo >= 4 ? yr : yr - 1;
    const fyEnd   = fyStart + 1;
    const fySuffix = String(fyStart).slice(-2) + '-' + String(fyEnd).slice(-2);
    const prefix   = 'CERT/' + fySuffix + '/';

    let maxSeq = 0;
    existingNos.forEach(function(no) {
      if (no.toUpperCase().startsWith(prefix.toUpperCase())) {
        const parts = no.split('/');
        const seq   = parseInt(parts[parts.length - 1], 10);
        if (!isNaN(seq) && seq > maxSeq) maxSeq = seq;
      }
    });

    const certNo = prefix + String(maxSeq + 1).padStart(3, '0');
    return jsonOut({ success: true, certNo: certNo });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

function handleGetCertificates() {
  try {
    const sheet   = getSheet(SHEET_CERTIFICATES);
    const data    = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonOut({ success: true, certificates: [] });
    const headers = data[0];
    const certificates = data.slice(1).map(function(row) {
      var obj = {};
      headers.forEach(function(h, i) { obj[h] = row[i] !== undefined ? row[i] : ''; });
      return obj;
    }).filter(function(c) { return c['Cert No']; });
    return jsonOut({ success: true, certificates: certificates });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

function handleSaveCertificate(body, username) {
  try {
    const sheet  = getSheet(SHEET_CERTIFICATES);
    const certNo = (body.certNo || '').toString().trim();

    if (!certNo) return jsonOut({ success: false, message: 'Certificate No. is required' });

    const allData = sheet.getDataRange().getValues();
    const exists  = allData.slice(1).some(function(r) {
      return (r[0] || '').toString().trim().toLowerCase() === certNo.toLowerCase();
    });
    if (exists) return jsonOut({ success: false, message: 'Duplicate certificate number — ' + certNo + ' already exists' });

    const now = new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' });

    sheet.appendRow([
      certNo,
      body.certType          || '',
      body.issueDate         || '',
      body.validUntil        || '',
      body.refNo             || '',
      body.site              || '',
      body.recipientName     || '',
      body.recipientAddress  || '',
      body.recipientPhone    || '',
      body.recipientEmail    || '',
      body.recipientGstin    || '',
      body.itemsJson         || '[]',
      body.remarks           || '',
      username,
      now,
      ''  // Certificate PDF URL — filled in lazily on first WhatsApp/Email send
    ]);

    return jsonOut({ success: true });

  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// buildCertificateHtml() ports the browser's printCertificateHtml() template
// (index.html) full-page flex layout — the same flex approach LedgerPrintGas
// already proved Drive's HTML-to-PDF converter handles fine — with only the
// two deviations already established for the invoice engine, both for the
// same underlying reason (Drive, not a real browser, renders this):
//   1. Stamp+signature overlap uses vertical-align:middle + a negative
//      margin instead of position:absolute/transform (which throws off
//      Drive's rendering elsewhere in this file — see LedgerPrintGas's
//      header comment).
//   2. The certificate's decorative double border is two nested plain divs
//      (each with its own solid border) instead of a position:absolute
//      ::before pseudo-element with an inset — same reasoning.
// Everything else — full A4 height, flex-column page, footer pinned to the
// bottom via margin-top:auto — mirrors the browser version exactly so the
// emailed/WhatsApp PDF looks the same as Print/Preview.
// ─────────────────────────────────────────────────────────────────────────

// HTML-escapes any user-supplied text before it's interpolated into the
// certificate document — mirrors the esc() helper already used throughout
// LedgerPrintGas. Lower urgency here than on the client (Drive's HTML-to-PDF
// conversion doesn't execute script in an authenticated browser session),
// but unescaped '<'/'>' can still corrupt the PDF's layout, so it's applied
// for the same reason and the same way.
function certEsc(v) {
  return String(v == null ? '' : v)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}

var CERT_MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
function certDmy(v) {
  if (!v) return '';
  var d = new Date(v);
  if (isNaN(d.getTime())) return String(v);
  return ('0' + d.getDate()).slice(-2) + '-' + CERT_MONTHS[d.getMonth()] + '-' + d.getFullYear();
}

      '</div>';

  // actually has something in it; otherwise it's dropped entirely (not

    '.cert-border{border:1px solid #E57373;border-radius:2px;flex:1;display:flex;flex-direction:column;padding:8mm 10mm;box-sizing:border-box;}' +
    '.cert-header{display:flex;align-items:center;gap:4mm;border-bottom:2px solid #B71C1C;padding-bottom:6mm;margin-bottom:6mm;}' +
    '.cert-header-logo{flex:0 0 24mm;width:24mm;display:flex;align-items:center;justify-content:center;}' +
    '.cert-header-logo img{max-width:24mm;max-height:20mm;object-fit:contain;}' +

    '.cert-title{font-size:22px;font-weight:bold;text-transform:uppercase;letter-spacing:3px;color:#2A2660;margin:5mm 0 2mm;}' +
    '.cert-type{font-size:12px;color:#555;letter-spacing:1px;text-transform:uppercase;}' +
    '.cert-body{flex:1;}' +

function renderCertificatePdf(data, fileName) {
  var html = buildCertificateHtml(data);
  var tempFile = null, pdfBlob;
  try {
    tempFile = DriveApp.createFile(Utilities.newBlob(html, 'text/html', 'temp_certificate.html'));
    pdfBlob = tempFile.getAs('application/pdf');
    pdfBlob.setName(fileName || ((data.certNo || 'Certificate') + '.pdf'));
  } finally {
    if (tempFile) { try { tempFile.setTrashed(true); } catch (e) { Logger.log('Temp file cleanup: ' + e); } }
  }
  return pdfBlob;
}

// Finds a certificate row by number and returns enough to both read and, if
// needed, write back to it (caching the generated PDF URL) — mirrors
// findInvoiceRow above.
function findCertificateRow(certNo) {
  const sheet = getSheet(SHEET_CERTIFICATES);
  const allData = sheet.getDataRange().getValues();
  const headers = allData[0].map(function(h) { return String(h).trim(); });
  for (var i = 1; i < allData.length; i++) {
    if ((allData[i][0] || '').toString().trim().toLowerCase() === certNo.toLowerCase()) {
      return { sheet: sheet, headers: headers, rowIdx: i, row: allData[i] };
    }
  }
  return null;
}

function buildCertificatePayloadFromRow(headers, row, certNo) {
  var c = {};
  headers.forEach(function(h, j) { c[h] = row[j]; });
  var items; try { items = JSON.parse(c['Items (JSON)'] || '[]'); } catch (e) { items = []; }
  return {
    certNo: c['Cert No'] || certNo, certType: c['Cert Type'] || '',
    issueDate: c['Issue Date'] || '', validUntil: c['Valid Until'] || '',
    refNo: c['Ref No'] || '', site: c['Site'] || '',
    recipientName: c['Recipient Name'] || '', recipientAddress: c['Recipient Address'] || '',
    recipientPhone: c['Recipient Phone'] || '', recipientEmail: c['Recipient Email'] || '',
    recipientGstin: c['Recipient GSTIN'] || '',
    items: items, remarks: c['Remarks'] || ''
  };
}

// Builds the certificate PDF and saves it to Drive, shared "anyone with the
// link" — the WhatsApp message's link must keep working after the request
// ends, so (unlike sendCertificateCopyToBuyer's temp-file-then-trash
// pattern) this copy persists. Mirrors saveInvoicePdfToDrive above.
function saveCertificatePdfToDrive(data) {
  const safeCertNo = (data.certNo || 'Certificate').replace(/\//g, '-');
  const safeName = String(data.recipientName || 'Recipient').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName = safeName + ' - ' + safeCertNo + '.pdf';

  const pdfBlob = renderCertificatePdf(data, pdfFileName);

  const folder = getOrCreateAppFolder('GME InvoiceApp - Shared Certificate PDFs');
  const pdfFile = folder.createFile(pdfBlob);
  pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return pdfFile.getUrl();
}

// Emails the certificate PDF straight to the recipient. Mirrors
// sendInvoiceCopyToBuyer above.
function sendCertificateCopyToBuyer(data) {
  const toEmail = (data.recipientEmail || '').toString().trim();
  if (!toEmail) return { sent: false, message: 'No recipient email on file for this certificate.' };

  const safeCertNo = (data.certNo || 'Certificate').replace(/\//g, '-');
  const safeName = String(data.recipientName || 'Recipient').replace(/[^a-zA-Z0-9 ]/g, '').trim();
  const pdfFileName = safeName + ' - ' + safeCertNo + '.pdf';

  const pdfBlob = renderCertificatePdf(data, pdfFileName);

  const subject = 'Certificate ' + (data.certNo || '') + ' from ' + (COMPANY.name || '');
  const emailBody =
    'Dear ' + (data.recipientName || 'Customer') + ',\n\n' +
    'Please find your certificate attached.\n\n' +
    '  Certificate No. : ' + (data.certNo || '-') + '\n' +
    (data.validUntil ? '  Valid Until     : ' + certDmy(data.validUntil) + '\n' : '') +
    '\nThank you.\n\n' +
    'Regards,\n' + (COMPANY.name || '');

  MailApp.sendEmail(toEmail, subject, emailBody, {
    name: COMPANY.name || 'Certificate',
    attachments: [pdfBlob]
  });

  return { sent: true, message: 'Certificate emailed to ' + toEmail };
}

function handleSendCertificateToBuyer(body) {
  const certNo = (body.certNo || '').toString().trim();
  if (!certNo) return jsonOut({ success: false, message: 'Certificate No. is required' });
  const found = findCertificateRow(certNo);
  if (!found) return jsonOut({ success: false, message: 'Certificate not found: ' + certNo });
  const payload = buildCertificatePayloadFromRow(found.headers, found.row, certNo);
  try {
    const result = sendCertificateCopyToBuyer(payload);
    return jsonOut({ success: result.sent, message: result.message });
  } catch (err) {
    return jsonOut({ success: false, message: err.toString() });
  }
}

// Returns { phone, recipientName, pdfUrl } for the frontend to build a
// wa.me click-to-chat link from. Regenerates the PDF fresh on every call
// (see the comment inside handleGetInvoiceWhatsappLink above) and
// overwrites the cached "Certificate PDF URL" column with the new link.
function handleGetCertificateWhatsappLink(body) {
  const certNo = (body.certNo || '').toString().trim();
  if (!certNo) return jsonOut({ success: false, message: 'Certificate No. is required' });
  const found = findCertificateRow(certNo);
  if (!found) return jsonOut({ success: false, message: 'Certificate not found: ' + certNo });

  const phoneCol = found.headers.indexOf('Recipient Phone');
  const phone = (phoneCol > -1 ? found.row[phoneCol] : '').toString().trim();
  if (!phone) return jsonOut({ success: false, message: 'No recipient phone on file for this certificate.' });

  // Always regenerate — see the matching comment in
  // handleGetInvoiceWhatsappLink above for why a cached URL is unsafe here.
  const urlCol = found.headers.indexOf('Certificate PDF URL');
  var pdfUrl;
  const payload = buildCertificatePayloadFromRow(found.headers, found.row, certNo);
  try {
    pdfUrl = saveCertificatePdfToDrive(payload);
    if (urlCol > -1) found.sheet.getRange(found.rowIdx + 1, urlCol + 1).setValue(pdfUrl);
  } catch (err) {
    return jsonOut({ success: false, message: 'Could not generate certificate PDF: ' + err.toString() });
  }

  const nameCol = found.headers.indexOf('Recipient Name');
  return jsonOut({
    success: true,
    phone: phone,
    recipientName: nameCol > -1 ? (found.row[nameCol] || '') : '',
    pdfUrl: pdfUrl
  });
}
